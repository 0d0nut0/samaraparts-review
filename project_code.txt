"""
Telegram-бот Parts.Site на базе сайта: python -m app.bot

Нужны TELEGRAM_BOT_TOKEN, TELEGRAM_BOT_USERNAME и SITE_URL (https — для Mini App) в .env,
та же база и Redis, что у сайта. Запускается одним процессом рядом с сайтом (docker-compose: сервис bot).
"""
import asyncio
import logging
import sys

from aiogram import Dispatcher
from aiogram.fsm.storage.memory import MemoryStorage
from aiogram.fsm.storage.redis import RedisStorage
from aiogram.types import BotCommand, MenuButtonWebApp, WebAppInfo

from app.bot import shop, staff
from app.bot.common import AccountMiddleware, webapp_url
from app.config import settings
from app.services import telegram

logging.basicConfig(level=logging.INFO, format="%(asctime)s [%(levelname)s] %(name)s: %(message)s")
logger = logging.getLogger("bot")


def build_dispatcher(storage=None) -> Dispatcher:
    dp = Dispatcher(storage=storage or MemoryStorage())
    dp.update.outer_middleware(AccountMiddleware())
    dp.include_router(staff.router)  # действия администратора раньше «текст = поиск»
    dp.include_router(shop.router)
    return dp


async def main() -> None:
    if not settings.TELEGRAM_BOT_TOKEN:
        sys.exit("TELEGRAM_BOT_TOKEN не задан в .env")
    bot = telegram.get_bot()
    # Состояния диалогов (оформление, переписка) переживают перезапуск бота
    dp = build_dispatcher(RedisStorage.from_url(settings.REDIS_URL))
    await bot.set_my_commands([
        BotCommand(command="start", description="Главное меню"),
        BotCommand(command="catalog", description="Каталог"),
        BotCommand(command="search", description="Поиск по модели и детали"),
        BotCommand(command="cart", description="Корзина"),
        BotCommand(command="orders", description="Мои заказы"),
        BotCommand(command="profile", description="Профиль и оптовые цены"),
    ])
    url = webapp_url()
    if url:
        await bot.set_chat_menu_button(menu_button=MenuButtonWebApp(text="Магазин", web_app=WebAppInfo(url=url)))
    else:
        logger.warning("SITE_URL не https — кнопка Mini App выключена (Telegram открывает только https)")
    me = await bot.get_me()
    logger.info("Бот @%s запущен, сайт %s", me.username, settings.SITE_URL)
    await dp.start_polling(bot)


if __name__ == "__main__":
    asyncio.run(main(), loop_factory=asyncio.SelectorEventLoop if sys.platform == "win32" else None)

"""Общее для обработчиков бота: сессия БД и аккаунт в каждом обновлении, клавиатуры, форматирование."""
import html
from typing import Any, Awaitable, Callable

)

BTN_CATALOG = "🛍 Каталог"
BTN_SEARCH = "🔍 Поиск"
BTN_CART = "🛒 Корзина"
BTN_ORDERS = "📦 Мои заказы"
BTN_CHAT = "💬 Менеджер"
BTN_PROFILE = "👤 Профиль"
BTN_ADMIN = "⚙️ Админка"


class AccountMiddleware(BaseMiddleware):
    """Каждому обновлению — своя сессия БД и аккаунт сайта этого Telegram (создаётся при первом обращении)."""

    async def __call__(self, handler: Callable[[TelegramObject, dict], Awaitable[Any]], event: TelegramObject, data: dict) -> Any:
        tg = data.get("event_from_user")
        async with AsyncSessionLocal() as db:
            data["db"] = db
            if tg and not tg.is_bot:
                data["user"] = await accounts.telegram_user(db, tg.id, tg.full_name, tg.username or "")
            return await handler(event, data)


def webapp_url(path: str = "/tg/app") -> str | None:
    """Mini App открывается только по https (требование Telegram)."""
    return site(path) if settings.SITE_URL.startswith("https://") else None


def main_menu(user) -> ReplyKeyboardMarkup:
    url = webapp_url()
    catalog = KeyboardButton(text=BTN_CATALOG, web_app=WebAppInfo(url=url)) if url else KeyboardButton(text=BTN_CATALOG)
    rows = [
        [catalog, KeyboardButton(text=BTN_SEARCH)],
        [KeyboardButton(text=BTN_CART), KeyboardButton(text=BTN_ORDERS)],
        [KeyboardButton(text=BTN_CHAT), KeyboardButton(text=BTN_PROFILE)],
    ]
    if user is not None and user.is_superuser:
        rows.append([KeyboardButton(text=BTN_ADMIN)])
    return ReplyKeyboardMarkup(keyboard=rows, resize_keyboard=True, input_field_placeholder="Модель и деталь: «iPhone 11 дисплей»")


def kb(rows: list[list[tuple[str, str]]]) -> InlineKeyboardMarkup | None:
    """[[(текст, callback | "url:…")]] → inline-клавиатура (ссылки не на https Telegram не принимает — пропускаем)."""
    from app.services.telegram import keyboard
    return keyboard(rows)


def esc(text) -> str:
    return html.escape(str(text or ""))

# Вход и привязка — только после подтверждения кнопкой. Иначе ссылку, созданную злоумышленником на своём устройстве,
# можно прислать жертве: она нажмёт «Запустить», и злоумышленник войдёт в её аккаунт (или привяжет её Telegram к своему).
WARNING = "⚠️ Если вы сейчас не нажимали эту кнопку на сайте сами — нажмите «Отмена»: кто-то пытается получить доступ."


async def _login_confirm(message: Message, token: str, user: User):
    raw = telegram.peek_token("login", token)
    if not raw or not raw.startswith("?"):
        await message.answer("Ссылка для входа устарела. Нажмите «Войти через Telegram» на сайте ещё раз.",
                             reply_markup=main_menu(user))
        return
    try:
        info = json.loads(raw[1:])
    except ValueError:
        info = {}
    await message.answer(
        f"🔐 <b>Вход на сайт Parts.Site</b> в аккаунт <b>{esc(user.full_name or user.username)}</b>.\n"
        f"Запрос с IP <code>{esc(info.get('ip', '?'))}</code>, браузер: {esc((info.get('ua') or '?')[:80])}\n\n{WARNING}",
        reply_markup=kb([[("✅ Это я — войти", f"lg:ok:{token}"), ("Отмена", f"lg:no:{token}")]]),
    )


@router.callback_query(F.data.regexp(r"^lg:(ok|no):[A-Za-z0-9]+$"))
async def login_decision(cb: CallbackQuery, user: User):
    _, action, token = cb.data.split(":")
    raw = telegram.peek_token("login", token)
    if not raw or not raw.startswith("?"):
        await cb.message.edit_text("Ссылка для входа устарела.")
        await cb.answer()
        return
    if action == "ok":
        telegram.set_token("login", token, user.id)
        await cb.message.edit_text("✅ Готово — вы вошли на сайте. Вернитесь в браузер, страница обновится сама.")
    else:
        telegram.take_token("login", token)
        await cb.message.edit_text("Вход отменён.")
    await cb.answer()


async def _link_confirm(message: Message, token: str, db: AsyncSession):
    target_id = telegram.peek_token("link", token)
    target = (await db.execute(select(User).where(User.id == int(target_id)))).scalar_one_or_none() if target_id else None
    if not target:
        await message.answer("Ссылка для привязки устарела. Нажмите «Привязать Telegram» в личном кабинете ещё раз.")
        return
    await message.answer(
        f"🔗 <b>Привязать этот Telegram к аккаунту «{esc(target.username)}» на сайте?</b>\n"
        "Заказы и переписка из бота перейдут в этот аккаунт, а войти в него можно будет через Telegram.\n\n" + WARNING,
        reply_markup=kb([[("✅ Привязать", f"lk:ok:{token}"), ("Отмена", f"lk:no:{token}")]]),
    )


@router.callback_query(F.data.regexp(r"^lk:(ok|no):[A-Za-z0-9]+$"))
async def link_decision(cb: CallbackQuery, db: AsyncSession, user: User):
    _, action, token = cb.data.split(":")
    if action == "no":
        telegram.take_token("link", token)
        await cb.message.edit_text("Привязка отменена.")
        await cb.answer()
        return
    await cb.message.edit_reply_markup(reply_markup=None)
    await _link(cb.message, token, db, user, cb.from_user)
    await cb.answer()


async def _link(message: Message, token: str, db: AsyncSession, user: User, tg_user=None):
    """Привязка Telegram к аккаунту сайта (ссылка из личного кабинета)."""
    target_id = telegram.peek_token("link", token)
    target = (await db.execute(select(User).where(User.id == int(target_id)))).scalar_one_or_none() if target_id else None
    if not target:
        await message.answer("Ссылка для привязки устарела. Нажмите «Привязать Telegram» в личном кабинете ещё раз.")
        return
    try:
        tg_user = tg_user or message.from_user
        await accounts.link_telegram(db, target, tg_user.id, tg_user.full_name)
    except accounts.AccountError as e:
        await message.answer(f"❌ {esc(e)}")
        return
    telegram.take_token("link", token)
    await db.refresh(target)
    await message.answer(f"✅ Telegram привязан к аккаунту <b>{esc(target.username)}</b>. Сюда будут приходить статусы заказов "
                         "и ответы менеджера, а пароль можно восстановить через бота.", reply_markup=main_menu(target))

    tree = await catalog_tree.get_tree(db)
    rows = [[(f"{n.icon or ''} {n.name} · {n.count}".strip(), f"c:{n.id}:0")] for n in tree.sections]
    url = webapp_url()
    if url:

    return "📂 <b>Каталог</b>\nВыберите раздел или просто напишите, что ищете.", kb(rows)


@router.callback_query(F.data.startswith("c:"))
async def category(cb: CallbackQuery, db: AsyncSession, user: User):
    _, cid, page = cb.data.split(":")
    tree = await catalog_tree.get_tree(db)
    node = tree.by_id.get(int(cid))
    if not node:
        await cb.answer("Раздел не найден", show_alert=True)
        return
    back = f"c:{node.parent_id}:0" if node.parent_id else "sec"
    if node.children and int(page) == 0:
        rows = [[(f"{c.name} · {c.count}", f"c:{c.id}:0")] for c in node.children if c.count]
        rows.append([("Все товары раздела", f"cl:{node.id}:0")])
        rows.append([("← Назад", back)])
        await cb.message.edit_text(f"📂 <b>{esc(node.name)}</b>", reply_markup=kb(rows))
        await cb.answer()
        return
    await _category_list(cb, db, user, node, 0, back)

    tree = await catalog_tree.get_tree(db)
    cond = [Product.category_id.in_(tree.descendant_ids(node))]

async def _product_list(db, user, cond, order, page, title, page_prefix, back=None):
    # Товары с карточкой склада и не скрытые — те же, что видит витрина
    cond = [*cond, Product.is_active == 1]
    total = (await db.execute(select(func.count()).select_from(MobaPrice).join(Product, Product.sku == MobaPrice.sku)
                              .where(*cond))).scalar_one()
    rows_db = (
        await db.execute(
            select(MobaPrice, Product.id).join(Product, Product.sku == MobaPrice.sku)
            .where(*cond).order_by(*order).offset(page * PAGE).limit(PAGE)
        )
    ).all()
    stocks = await _stock_map(db, [mp.sku for mp, _ in rows_db], None)
    rows = []
    for mp, pid in rows_db:
        price = buyer_prices(mp, user.is_wholesale)["price"]
        st = stocks.get(mp.sku)
        mark = "🟢" if st and st.total > 0 else "⚪️"
        rows.append([(f"{mark} {mp.name[:40]} — {money(price) if price is not None else 'цена по запросу'}", f"p:{pid}")])
    nav = []
    if page > 0:
        nav.append(("◀", f"{page_prefix}:{page - 1}"))
    pages = max(1, (total + PAGE - 1) // PAGE)
    if pages > 1:
        nav.append((f"{page + 1}/{pages}", "noop"))
    if page + 1 < pages:
        nav.append(("▶", f"{page_prefix}:{page + 1}"))
    rows.append(nav)
    if back:
        rows.append([("← Назад", back)])
    head = f"{title}\nНайдено: {total}. 🟢 — есть на складе." if total else f"{title}\nНичего не нашлось. Напишите менеджеру — подберём."
    if not total:
        rows.append([("💬 Спросить менеджера", "chat:me:reply")])
    return head, kb(rows)

async def _product_card(message: Message, db: AsyncSession, user: User, pid: int):
    row = (await db.execute(select(Product, MobaPrice).join(MobaPrice, MobaPrice.sku == Product.sku)
                            .where(Product.id == pid, Product.is_active == 1))).first()
    if not row:
        await message.answer("Товар не найден или снят с продажи.")
        return
    product, mp = row
    bp = buyer_prices(mp, user.is_wholesale)
    stocks = (await db.execute(select(City.name, Stock.quantity).join(Stock, Stock.city_id == City.id)
                               .where(Stock.product_id == pid).order_by(City.id))).all()
    stock_text = "\n".join(f"• {esc(c)}: {'<b>' + str(q) + ' шт.</b>' if q > 0 else 'нет'}" for c, q in stocks) or "• нет на складе"
    if bp["price"] is None:
        price_text = "💰 <b>Цена по запросу</b> — менеджер назовёт цену и зарезервирует товар."
    else:
        price_text = (f"💰 <b>{money(bp['price'])}</b>{' (опт)' if bp['wholesale'] else ''}\n"
                      f"картой при получении — {money(bp['card'])}")
    text = f"<b>{esc(mp.name)}</b>\n<code>{esc(mp.sku)}</code>\n\n{price_text}\n\n📦 Наличие:\n{stock_text}"
    in_stock = any(q > 0 for _, q in stocks)
    rows = []
    if bp["price"] is not None and in_stock:
        rows.append([("🛒 В корзину", f"add:{pid}")])
    rows.append([("💬 Спросить менеджера", f"ask:{pid}"), ("На сайте", "url:" + site(f"/catalog/{mp.sku}"))])
    if mp.image and mp.image.startswith("http") and len(text) <= 1024:
        try:
            await message.answer_photo(mp.image, caption=text, reply_markup=kb(rows))
            return
        except Exception:  # Telegram не смог скачать картинку — покажем карточку без фото
            logger.info("Фото товара %s не отправилось", mp.sku)
    await message.answer(text, reply_markup=kb(rows))


@router.callback_query(F.data.startswith("add:"))
async def add(cb: CallbackQuery, db: AsyncSession, user: User):
    pid = int(cb.data[4:])
    sku = (await db.execute(select(Product.sku).where(Product.id == pid))).scalar_one_or_none()
    qty = await cart.user_add(db, user.id, sku, 1) if sku else 0
    if not qty:
        await cb.answer("Этот товар сейчас нельзя купить — спросите менеджера", show_alert=True)
        return
    count = await cart.user_count(db, user.id)
    await cb.answer(f"Добавлено, в корзине {qty} шт. этого товара (всего {count})")

async def _cart(db: AsyncSession, user: User):
    items = await cart.user_cart_items(db, user.id, user.is_wholesale)
    if not items:
        return "🛒 Корзина пуста. Напишите, что ищете, или откройте каталог.", kb([[("📂 Каталог", "sec")]])
    pids = dict((await db.execute(select(Product.sku, Product.id).where(Product.sku.in_([i["sku"] for i in items])))).all())
    totals = cart.cart_totals(items)
    lines, rows = [], []
    for n, i in enumerate(items, 1):
        lines.append(f"{n}. {esc(i['name'])}\n    {i['qty']} × {money(i['price'])} = <b>{money(i['sum'])}</b>")
        pid = pids[i["sku"]]
        rows.append([(f"{n}.", "noop"), ("➖", f"dec:{pid}"), (f"{i['qty']} шт.", "noop"), ("➕", f"inc:{pid}"), ("🗑", f"del:{pid}")])
    rows.append([("✅ Оформить заказ", "co")])
    rows.append([("Очистить", "clear")])
    opt = " (оптовые цены)" if user.is_wholesale else ""
    text = ("🛒 <b>Корзина</b>" + opt + "\n\n" + "\n".join(lines) +
            f"\n\nИтого: <b>{money(totals['total'])}</b>\nкартой при получении — {money(totals['total_card'])}")
    return text, kb(rows)


@router.callback_query(F.data.regexp(r"^(inc|dec|del):\d+$"))
async def cart_change(cb: CallbackQuery, db: AsyncSession, user: User):
    action, pid = cb.data.split(":")
    sku = (await db.execute(select(Product.sku).where(Product.id == int(pid)))).scalar_one_or_none()
    items = {i["sku"]: i["qty"] for i in await cart.user_cart_items(db, user.id, user.is_wholesale)}
    if sku in items:
        qty = {"inc": items[sku] + 1, "dec": items[sku] - 1, "del": 0}[action]
        await cart.user_set(db, user.id, sku, qty)
    text, markup = await _cart(db, user)
    await cb.message.edit_text(text, reply_markup=markup)
    await cb.answer()

async def checkout_start(cb: CallbackQuery, db: AsyncSession, user: User, state: FSMContext):
    items = await cart.user_cart_items(db, user.id, user.is_wholesale)
    if not items:
        await cb.answer("Корзина пуста", show_alert=True)
        return
    # Где можно собрать весь заказ — как на сайте при оформлении
    need = {i["sku"]: i["qty"] for i in items}
    cities = (await db.execute(select(City).where(City.is_active.is_(True)).order_by(City.id))).scalars().all()

                              .where(Product.sku.in_(list(need))))).all()
    short = {c.id: len(need) for c in cities}
    for sku, city_id, qty in stock:
        if city_id in short and qty >= need[sku]:
            short[city_id] -= 1
    rows = [[(f"{c.name} — {'всё в наличии' if short[c.id] == 0 else 'не хватает ' + str(short[c.id]) + ' поз.'}", f"co:city:{c.id}")]
            for c in cities]
    await state.update_data(co={})
    await cb.message.answer("📍 <b>Шаг 1 из 3.</b> Где заберёте заказ? Товар резервируется на складе этого города.", reply_markup=kb(rows))
    await cb.answer()

@router.callback_query(F.data == "co:ok")

async def _checkout_ok(cb: CallbackQuery, db: AsyncSession, user: User, state: FSMContext):
    data = await state.get_data()
    if not data.get("city_id") or not data.get("phone"):
        await cb.answer("Начните оформление заново из корзины", show_alert=True)
        return
    items = await cart.user_cart_items(db, user.id, user.is_wholesale)
    if not items:
        await cb.answer("Корзина пуста", show_alert=True)
        return
    try:
        order = await order_service.create_order(
            db=db, user_id=user.id, city_id=data["city_id"],
            cart_items=[{"sku": i["sku"], "qty": i["qty"], "price": i["price"]} for i in items],
            customer_name=(user.full_name or cb.from_user.full_name or user.username)[:200],
            customer_phone=data["phone"], wholesale=user.is_wholesale,
            comment=data.get("comment") or None, source="bot",
        )
        await cart.user_clear(db, user.id, commit=False)
        await db.commit()
    except order_service.OrderError as e:
        await cb.message.answer(f"❌ {esc(e)}")
        await cb.answer()
        return
    await state.clear()
    await cb.message.edit_reply_markup(reply_markup=None)
    await cb.answer("Заказ оформлен")
    await telegram.notify_new_order(order.id)  # покупателю — подтверждение, администраторам — уведомление

async def _own_thread(db: AsyncSession, user: User, create: bool) -> ChatThread | None:

    thread = ChatThread(user_id=user.id, name=user.full_name, phone=user.phone)

    return thread

@router.callback_query(F.data.regexp(r"^ord:\d+:[A-Z]+$"))
async def order_status(cb: CallbackQuery, db: AsyncSession, user: User):
    if not user.is_superuser:
        await _deny(cb)
        return
    _, order_id, status = cb.data.split(":")
    if status not in OrderStatus.__members__:
        await cb.answer("Неизвестный статус", show_alert=True)
        return
    current = await db.get(Order, int(order_id))
    if current and current.status.name == status:
        await cb.answer("Статус уже установлен")
        return
    try:
        order = await order_service.change_order_status(db, int(order_id), OrderStatus[status], user.id)
        await db.commit()
    except order_service.OrderError as e:
        await cb.answer(str(e), show_alert=True)
        return
    label = telegram.ORDER_STATUS_TEXT[status]
    await cb.answer(f"Заказ № {order_id}: {label}")
    try:
        await cb.message.edit_text(cb.message.html_text + f"\n\n➡️ <b>{label}</b> — {esc(user.full_name or user.username)}",
                                   reply_markup=telegram.keyboard(telegram.order_admin_rows(order)))
    except Exception:
        pass
    await telegram.notify_order_status(int(order_id))


@router.callback_query(F.data.regexp(r"^ws:\d+:(APPROVED|REJECTED)$"))
async def wholesale_decision(cb: CallbackQuery, db: AsyncSession, user: User):
    if not user.is_superuser:
        await _deny(cb)
        return
    _, app_id, status = cb.data.split(":")
    application = (await db.execute(select(WholesaleApplication).options(joinedload(WholesaleApplication.user))
                                    .where(WholesaleApplication.id == int(app_id)))).scalar_one_or_none()
    if not application or application.status != ApplicationStatus.PENDING:
        await cb.answer("Заявка уже обработана", show_alert=True)
        return
    application.status = ApplicationStatus[status]
    approved = status == "APPROVED"
    if approved and application.user.role == UserRole.RETAIL:
        application.user.role = UserRole.WHOLESALE
    await db.commit()
    await cb.answer("Одобрено" if approved else "Отклонено")
    try:
        await cb.message.edit_text(cb.message.html_text + f"\n\n➡️ <b>{'Одобрено' if approved else 'Отклонено'}</b>", reply_markup=None)
    except Exception:
        pass
    await telegram.notify_wholesale_result(application.user_id, approved)

@app.middleware("http")
async def add_security_headers(request: Request, call_next):
    """
    Добавляем HTTP-заголовки безопасности (OWASP рекомендации).
    """
    response = await call_next(request)
    response.headers["X-Content-Type-Options"] = "nosniff"
    response.headers["X-XSS-Protection"] = "1; mode=block"
    if HTTPS_ONLY:
        # HSTS имеет смысл только за HTTPS, иначе на localhost браузер «залипнет» на https
        response.headers["Strict-Transport-Security"] = "max-age=31536000; includeSubDomains"
    # Всё своё (Tailwind, HTMX, шрифты — /static). telegram.org — скрипт Mini App (/tg/app).
    # Встраивать сайт во фрейм можно только Telegram Web (Mini App в браузерной версии Telegram);
    # frame-ancestors заменяет X-Frame-Options.
    response.headers["Content-Security-Policy"] = (
        "default-src 'self'; "
        "style-src 'self' 'unsafe-inline'; "
        "font-src 'self'; "
        "script-src 'self' 'unsafe-inline' 'unsafe-eval' https://telegram.org; "
        "img-src 'self' data: https:; "
        "connect-src 'self'; "
        "frame-ancestors 'self' https://web.telegram.org https://*.telegram.org;"
    )
    response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
    response.headers["Permissions-Policy"] = "camera=(), microphone=(), geolocation=(), payment=()"
    if request.url.path.startswith(("/admin", "/auth", "/checkout", "/cart", "/chat", "/tg")):
        response.headers["Cache-Control"] = "no-store"  # личные страницы не кэшируются браузером и прокси
    return response

@app.middleware("http")
async def add_user_to_request(request: Request, call_next):
    """
    Инжектируем текущего пользователя в request.state.user перед каждым запросом.
    Шаблоны могут читать {{ request.state.user }} без дополнительных зависимостей.
    """
    request.state.user = None
    user_id = request.session.get("user_id")

    async with AsyncSessionLocal() as db:
        if user_id:
            result = await db.execute(select(User).where(User.id == user_id))
            request.state.user = result.scalar_one_or_none()
            if request.state.user is not None and request.session.get("pwv") != accounts.session_fingerprint(request.state.user):
                request.state.user = None  # пароль сменили после входа (или сессия до этой проверки) — входим заново
            if request.state.user is None:
                # Пользователь удалён — забываем его в сессии, иначе корзина пишет в несуществующий аккаунт
                request.session.pop("user_id", None)
        # Город покупателя (выбор в шапке): по нему показываем наличие и предлагаем склад при заказе
        cities = await _active_cities(db)
    request.state.cities = cities
    request.state.city = next((c for c in cities if c.id == request.session.get("city_id")), cities[0] if cities else None)

    return await call_next(request)


_cities_cache: tuple[float, list] = (0.0, [])


async def _active_cities(db) -> list:
    """Активные города с кэшем на 5 минут — список нужен в шапке на каждой странице."""
    global _cities_cache
    if time.monotonic() - _cities_cache[0] > 300:
        rows = (await db.execute(select(City).where(City.is_active.is_(True)).order_by(City.id))).scalars().all()
        _cities_cache = (time.monotonic(), [SimpleNamespace(id=c.id, name=c.name) for c in rows])
    return _cities_cache[1]

"""
Модель пользователя.
Хранит логин и хеш пароля (PBKDF2-SHA256, реализован в app/routers/auth.py).
"""

from sqlalchemy import BigInteger, Column, Integer, String, Boolean, Enum
from sqlalchemy.orm import relationship
import enum
from app.database import Base

class UserRole(str, enum.Enum):
    """ADMIN — доступ к админке; WHOLESALE и RETAIL — покупатели (оптовые и розничные цены)."""
    ADMIN = "admin"
    WHOLESALE = "wholesale"
    RETAIL = "retail"


class User(Base):
    __tablename__ = "users"

    id = Column(Integer, primary_key=True, index=True)

    # Уникальное имя пользователя — используется как логин
    username = Column(String, unique=True, index=True, nullable=False)

    # Хеш вида "pbkdf2_sha256$<соль>$<хеш>" — пароль никогда не хранится в открытом виде
    password_hash = Column(String, nullable=False)

    is_admin = Column(Boolean, default=False, nullable=False)
    role = Column(Enum(UserRole), default=UserRole.RETAIL, nullable=False)

    # Привязанный Telegram: вход и восстановление пароля через бота, уведомления о заказах.
    # Покупатель, пришедший из бота, получает аккаунт автоматически (username «tg<id>»).
    telegram_id = Column(BigInteger, unique=True, nullable=True, index=True)
    full_name = Column(String(200), nullable=True)
    phone = Column(String(50), nullable=True)

    @property
    def is_superuser(self) -> bool:
        """Полный доступ к админке: флаг is_admin или роль ADMIN."""
        return bool(self.is_admin) or self.role == UserRole.ADMIN

    @property
    def is_wholesale(self) -> bool:
        """Покупает по оптовым ценам."""
        return self.role == UserRole.WHOLESALE

    def __repr__(self):
        return f"<User id={self.id} username={self.username!r}>"

    orders = relationship("Order", back_populates="user")
    cart_items = relationship("UserCart", back_populates="user")

def require():
    """Зависимость: пользователь авторизован и является администратором."""
    async def dependency(request: Request, db: AsyncSession = Depends(get_db)) -> User:
        user_id = request.session.get("user_id")
        if not user_id:
            raise HTTPException(status_code=303, headers={"Location": "/auth/login"})
        user = (await db.execute(select(User).where(User.id == user_id))).scalar_one_or_none()
        if not user or not user.is_superuser:
            raise HTTPException(status_code=403, detail="Недостаточно прав для этого раздела")
        return user
    return dependency

    if msg:
        url += ("&" if "?" in url else "?") + "msg=" + quote(msg)
    if error:
        url += ("&" if "?" in url else "?") + "error=" + quote(error)
    return RedirectResponse(url=url, status_code=303)


def _decimal(value: str | None, field: str, allow_empty: bool = True) -> Decimal | None:
    if value is None or not value.strip():
        if allow_empty:
            return None
        raise ValueError(f"Поле «{field}» обязательно")
    try:
        d = Decimal(value.replace(",", ".").strip())
    except InvalidOperation:
        raise ValueError(f"Поле «{field}»: введите число")
    if d < 0:
        raise ValueError(f"Поле «{field}» не может быть отрицательным")
    return d.quantize(Decimal("0.01"))

    if not (await db.execute(select(Supplier.id).where(Supplier.id == sid))).first():

    return sid


def _pages(total: int, page: int) -> tuple[int, int]:
    pages = max(1, (total + PAGE_SIZE - 1) // PAGE_SIZE)
    return pages, min(max(page, 1), pages)

async def order_create(request: Request, db: AsyncSession = Depends(get_db), admin: User = Depends(require())):
    form = await request.form()
    url = "/admin/orders/new" + (f"?thread={form.get('thread')}" if form.get("thread") else
                                 f"?user={form.get('user_id')}" if form.get("user_id") else "")
    name, phone = (form.get("customer_name") or "").strip()[:200], (form.get("customer_phone") or "").strip()[:50]
    if not name or sum(ch.isdigit() for ch in phone) < 5:
        return back(url, error="Укажите имя и телефон покупателя")
    customer = await db.get(User, int(form["user_id"])) if (form.get("user_id") or "").isdigit() else None
    wholesale = customer.is_wholesale if customer else form.get("price_type") == "opt"
    try:
        city_id = int(form.get("city_id") or 0)
    except ValueError:
        city_id = 0
    if not await _check_city(db, city_id):
        return back(url, error="Выберите склад")
    wanted: dict[str, int] = {}
    for sku, qty in zip(form.getlist("sku"), form.getlist("qty")):
        sku = sku.strip()
        if not sku:
            continue
        try:
            n = int(qty)
        except ValueError:
            return back(url, error=f"Количество для {sku}: введите число")
        if n > 0:
            wanted[sku] = wanted.get(sku, 0) + min(n, 999)
    if not wanted:
        return back(url, error="Добавьте товары")
    prices = {mp.sku: mp for mp in (await db.execute(select(MobaPrice).where(MobaPrice.sku.in_(list(wanted))))).scalars()}
    items = []
    for sku, qty in wanted.items():
        mp = prices.get(sku)
        price = buyer_prices(mp, wholesale)["price"] if mp else None
        if price is None:
            return back(url, error=f"У товара {sku} нет цены — задайте её в карточке товара")
        items.append({"sku": sku, "qty": qty, "price": price})
    try:
        order = await order_service.create_order(
            db=db, user_id=customer.id if customer else None, city_id=city_id, cart_items=items,
            customer_name=name, customer_phone=phone, address=(form.get("address") or "").strip()[:500] or None,
            wholesale=wholesale, comment=(form.get("comment") or "").strip()[:1000] or None, source="admin",
        )
        order.manager_note = (form.get("manager_note") or "").strip()[:1000] or None
        if customer and form.get("clear_cart"):
            await cart.user_clear(db, customer.id, commit=False)
        thread = await db.get(ChatThread, int(form["thread"])) if (form.get("thread") or "").isdigit() else None
        if thread:
            await chat.add_message(db, thread, "staff", f"Оформил для вас заказ № {order.id} на {_rub(order.total_amount)}. "
                                                        "Менеджер свяжется для подтверждения.", author_id=admin.id)
        await db.commit()
    except order_service.OrderError as e:
        return back(url, error=str(e))
    telegram.fire(telegram.notify_new_order(order.id))
    return back(f"/admin/orders/{order.id}", msg=f"Заказ № {order.id} оформлен")

    return telegram.money(value).replace("\u00a0", " ")

@router.post("/products/new")
async def product_create(request: Request, db: AsyncSession = Depends(get_db),
                         admin: User = Depends(require())):
    form = await request.form()
    sku = (form.get("sku") or "").strip()
    name = (form.get("name") or "").strip()
    try:
        if not sku or not name:
            raise ValueError("Артикул и название обязательны")
        if (await db.execute(select(Product.id).where(Product.sku == sku))).first():
            raise ValueError(f"Товар с артикулом {sku} уже существует")
        base = _decimal(form.get("price_base"), "Розничная цена", allow_empty=False)
        opt = _decimal(form.get("price_opt"), "Оптовая цена") or base
        card = _decimal(form.get("price_card"), "Цена по карте") or base
        cost = _decimal(form.get("purchase_price"), "Себестоимость")
        threshold = int(form.get("critical_stock_threshold") or 5)
        if threshold < 0:
            raise ValueError("Порог не может быть отрицательным")
        category_id = int(form.get("category_id"))
        if not (await db.execute(select(Category.id).where(Category.id == category_id))).first():
            raise ValueError("Выберите категорию")
        supplier_id = await _supplier_id(db, form.get("supplier_id"))
    except (ValueError, TypeError) as e:
        return back("/admin/products/new", error=str(e) if str(e) else "Проверьте поля формы")

    slug = re.sub(r"[^a-z0-9]+", "-", sku.lower()).strip("-") or sku
    product = Product(
        sku=sku, name=name, slug=slug, category_id=category_id,
        description=(form.get("description") or "").strip() or None,
        price_base=base, price_card=card, price_opt=opt, purchase_price=cost,
        critical_stock_threshold=threshold, is_active=1, supplier_id=supplier_id,
    )
    db.add(product)
    # Витрина читает цены из moba_prices — создаём (или обновляем оставшуюся от парсера) запись с нашими ценами
    mp = (await db.execute(select(MobaPrice).where(MobaPrice.sku == sku))).scalar_one_or_none()
    if mp:
        mp.name, mp.our_base, mp.our_card, mp.our_opt, mp.url = name, base, card, opt, None
    else:
        db.add(MobaPrice(sku=sku, name=name, price=base, our_base=base, our_card=card, our_opt=opt))
    await db.flush()
    for city in (await db.execute(select(City))).scalars().all():
        db.add(Stock(product_id=product.id, city_id=city.id, quantity=0))
    await db.commit()
    catalog_tree.invalidate()
    return back(f"/admin/products/{product.id}", msg="Товар создан. Оприходуйте остаток через «Приход товара».")

@router.post("/products/receive")
async def admin_receive_product_post(
    article: str = Form(...), quantity: int = Form(...), city_id: int = Form(...),
    purchase_price: str = Form(""), supplier_id: str = Form(""), notes: str = Form(""),
    db: AsyncSession = Depends(get_db), admin: User = Depends(require()),
):
    product = (await db.execute(select(Product).where(Product.sku == article.strip()))).scalar_one_or_none()
    if not product:
        return back(f"/admin/products/receive?article={quote(article)}", error="Товар с таким артикулом не найден")
    if quantity <= 0:
        return back(f"/admin/products/receive?article={quote(article)}", error="Количество должно быть больше нуля")
    if quantity > 100_000:
        return back(f"/admin/products/receive?article={quote(article)}", error="Слишком большое количество")
    try:
        cost = _decimal(purchase_price, "Закупочная цена")

        if not await _check_city(db, city_id):
            raise ValueError("Выберите склад")
    except ValueError as e:
        return back(f"/admin/products/receive?article={quote(article)}", error=str(e))

    if cost is not None:
        product.purchase_price = cost
    if supplier:

    stock = (
        await db.execute(select(Stock).where(Stock.product_id == product.id, Stock.city_id == city_id).with_for_update())
    ).scalar_one_or_none()
    if stock:
        stock.quantity += quantity
    else:
        db.add(Stock(product_id=product.id, city_id=city_id, quantity=quantity))
    db.add(InventoryTransaction(
        product_id=product.id, user_id=admin.id, quantity=quantity, type=TransactionType.RECEIPT,
        to_city_id=city_id, supplier_id=supplier, unit_cost=cost, notes=notes.strip() or None,
    ))
    await db.commit()
    return back("/admin/inventory", msg=f"Оприходовано {quantity} шт. «{product.name}»")


def _read_rows(filename: str, data: bytes) -> list[list]:
    """Строки таблицы из .xlsx или .csv (разделитель «;» или «,», кодировка UTF-8 или Windows-1251)."""
    if filename.lower().endswith(".xlsx"):
        import io
        import openpyxl
        wb = openpyxl.load_workbook(io.BytesIO(data), read_only=True, data_only=True)
        return [list(r) for r in wb.worksheets[0].iter_rows(values_only=True)]
    import csv
    import io
    try:
        text = data.decode("utf-8-sig")
    except UnicodeDecodeError:
        text = data.decode("cp1251")
    dialect = csv.Sniffer().sniff(text[:2000], delimiters=";,\t")
    return list(csv.reader(io.StringIO(text), dialect))


@router.post("/products/receive/upload", response_class=HTMLResponse)
async def admin_receive_upload(request: Request, db: AsyncSession = Depends(get_db), admin: User = Depends(require())):
    """Приход поставки файлом: Артикул / Количество / Цена (закупка). Всё одной транзакцией."""
    form = await request.form()
    upload = form.get("file")
    ctx = await _product_form_context(db)
    errors: list[str] = []

    def done(ok=0, units=0):
        return render(request, "admin_receive.html", admin, article=None, cities=ctx["cities"],
                      suppliers=ctx["suppliers"], report={"ok": ok, "units": units, "errors": errors})

    try:
        city_id = int(form.get("city_id") or 0)
        supplier = await _supplier_id(db, form.get("supplier_id"))
        if not await _check_city(db, city_id):
            raise ValueError("Выберите склад")
        if not upload or not getattr(upload, "filename", ""):
            raise ValueError("Выберите файл")
        data = await upload.read()
        if len(data) > 10 * 1024 * 1024:
            raise ValueError("Файл больше 10 МБ")
        rows = [r for r in _read_rows(upload.filename, data) if any(c not in (None, "") for c in r)]
    except Exception as e:  # битый файл, неверный формат
        errors.append(str(e) or "Не удалось прочитать файл")
        return done()
    if not rows:
        errors.append("Файл пустой")
        return done()

    header = [str(c or "").strip().lower() for c in rows[0]]
    find = lambda *keys: next((i for i, h in enumerate(header) if any(k in h for k in keys)), None)
    i_sku, i_qty, i_cost = find("артикул", "sku", "код"), find("кол", "qty", "шт"), find("закуп", "цен", "price", "cost")
    if i_sku is None or i_qty is None:
        errors.append("Не нашли колонки «Артикул» и «Количество» в первой строке")
        return done()

    ok = units = 0
    for n, r in enumerate(rows[1:], start=2):
        sku = str(r[i_sku] or "").strip() if i_sku < len(r) else ""
        if not sku:
            continue
        try:
            qty = int(float(str(r[i_qty]).replace(",", ".")))
            cost = _decimal(str(r[i_cost]), "Цена") if i_cost is not None and i_cost < len(r) and r[i_cost] not in (None, "") else None
        except (TypeError, ValueError):
            errors.append(f"Строка {n} ({sku}): количество или цена — не число")
            continue
        if qty <= 0 or qty > 100_000:
            errors.append(f"Строка {n} ({sku}): количество {qty}")
            continue
        product = (await db.execute(select(Product).where(Product.sku == sku))).scalar_one_or_none()
        if not product:
            errors.append(f"Строка {n}: артикул {sku} не найден")
            continue
        if cost is not None:
            product.purchase_price = cost
        if supplier:
            product.supplier_id = supplier
        stock = (await db.execute(select(Stock).where(Stock.product_id == product.id, Stock.city_id == city_id)
                                  .with_for_update())).scalar_one_or_none()
        if stock:
            stock.quantity += qty
        else:
            db.add(Stock(product_id=product.id, city_id=city_id, quantity=qty))
        db.add(InventoryTransaction(product_id=product.id, user_id=admin.id, quantity=qty, type=TransactionType.RECEIPT,
                                    to_city_id=city_id, supplier_id=supplier, unit_cost=cost,
                                    notes=f"Поставка файлом {upload.filename[:80]}"))
        ok += 1
        units += qty
    await db.commit()
    return done(ok, units)

@router.post("/products/{product_id}")
async def product_update(product_id: int, request: Request, db: AsyncSession = Depends(get_db),
                         admin: User = Depends(require())):
    product = (await db.execute(select(Product).where(Product.id == product_id))).scalar_one_or_none()
    if not product:
        raise HTTPException(status_code=404, detail="Товар не найден")
    form = await request.form()
    url = f"/admin/products/{product_id}"
    try:
        name = (form.get("name") or "").strip()
        if not name:
            raise ValueError("Название не может быть пустым")
        base = _decimal(form.get("price_base"), "Розничная цена", allow_empty=False)
        opt = _decimal(form.get("price_opt"), "Оптовая цена") or base
        card = _decimal(form.get("price_card"), "Цена по карте") or base
        cost = _decimal(form.get("purchase_price"), "Себестоимость")
        threshold = int(form.get("critical_stock_threshold") or 0)
        if threshold < 0:
            raise ValueError("Порог не может быть отрицательным")
        new_stocks = {int(k[6:]): int(v) for k, v in form.items() if k.startswith("stock_") and str(v).strip() != ""}
        if any(v < 0 for v in new_stocks.values()):
            raise ValueError("Остаток не может быть отрицательным")
        city_ids = set((await db.execute(select(City.id))).scalars())
        new_stocks = {cid: v for cid, v in new_stocks.items() if cid in city_ids}
        supplier_id = await _supplier_id(db, form.get("supplier_id"))
        category_id = int(form.get("category_id") or product.category_id)
        if not (await db.execute(select(Category.id).where(Category.id == category_id))).first():
            raise ValueError("Выберите категорию")
    except (ValueError, TypeError) as e:
        return back(url, error=str(e) or "Проверьте поля формы")

    product.name = name
    product.description = (form.get("description") or "").strip() or None
    product.category_id = category_id
    product.supplier_id = supplier_id
    product.price_base, product.price_opt, product.price_card = base, opt, card
    product.purchase_price = cost
    product.critical_stock_threshold = threshold

    mp = (await db.execute(select(MobaPrice).where(MobaPrice.sku == product.sku))).scalars().first()
    if mp:
        mp.name, mp.our_base, mp.our_opt, mp.our_card = name, base, opt, card
    else:
        db.add(MobaPrice(sku=product.sku, name=name, price=base, our_base=base, our_card=card, our_opt=opt))

    # Ручная корректировка остатков по городам (как «Изменить остаток вручную» в боте)
    current = {s.city_id: s for s in (await db.execute(select(Stock).where(Stock.product_id == product_id))).scalars()}
    for city_id, qty in new_stocks.items():
        stock = current.get(city_id)
        old = stock.quantity if stock else 0
        if qty == old:
            continue
        if stock:
            stock.quantity = qty
        else:
            db.add(Stock(product_id=product_id, city_id=city_id, quantity=qty))
        db.add(InventoryTransaction(
            product_id=product_id, user_id=admin.id, type=TransactionType.ADJUSTMENT, quantity=qty - old,
            to_city_id=city_id if qty > old else None, from_city_id=city_id if qty < old else None,
            notes=f"Корректировка из админки ({old} → {qty})",
        ))
    await db.commit()
    catalog_tree.invalidate()
    return back(url, msg="Изменения сохранены")

@router.post("/settings")
async def update_settings(request: Request, db: AsyncSession = Depends(get_db), admin: User = Depends(require())):
    form = await request.form()
    try:
        values = {}
        for key in PRICE_SETTINGS:
            raw = (form.get(key) or "").strip()
            if key == "BASE_FROM":
                if raw not in ("regular", "discount"):
                    raise ValueError("Источник цены: regular или discount")
            else:
                if Decimal(raw.replace(",", ".")) <= 0:
                    raise ValueError(f"{key} должен быть больше нуля")
                raw = raw.replace(",", ".")
            values[key] = raw
    except (InvalidOperation, ValueError) as e:
        return back("/admin/settings", error=str(e) or "Введите числа")

    for key, value in values.items():
        config = (await db.execute(select(SystemConfig).where(SystemConfig.key == key))).scalar_one_or_none()
        if config:
            config.value = value
        else:
            db.add(SystemConfig(key=key, value=value, description=PRICE_SETTINGS[key]))
    await db.commit()
    invalidate_config_cache()

    if form.get("recalculate"):
        updated = await recalculate_all(db)
        return back("/admin/settings", msg=f"Настройки сохранены, пересчитано цен: {updated}")
    return back("/admin/settings", msg="Настройки сохранены")

def _hash_password(password: str, salt: str | None = None) -> str:
    """
    Хешируем пароль через PBKDF2-SHA256.
    Формат итоговой строки: pbkdf2_sha256$<соль>$<хеш>.
    """
    if salt is None:
        salt = os.urandom(16).hex()
    key = hashlib.pbkdf2_hmac("sha256", password.encode(), salt.encode(), 260_000)
    return f"pbkdf2_sha256${salt}${key.hex()}"


def verify_password(plain: str, stored: str) -> bool:
    """
    Проверяем пароль, пересчитывая хеш с той же солью.
    Возвращает True, если пароли совпадают.
    """
    try:
        _, salt, _ = stored.split("$")
    except ValueError:
        return False
    return hmac.compare_digest(_hash_password(plain, salt), stored)


def get_password_hash(password: str) -> str:
    """Создаёт хеш пароля с рандомной солью."""
    return _hash_password(password)

async def get_current_user(request: Request, db: AsyncSession = Depends(get_db)) -> User | None:
    """
    Читает user_id из cookie-сессии и возвращает объект пользователя.
    Если сессии нет или пользователь не найден — вернёт None.
    """
    user_id = request.session.get("user_id")
    if not user_id:
        return None
    result = await db.execute(select(User).where(User.id == user_id))
    return result.scalar_one_or_none()


async def _merge_guest_cart(db: AsyncSession, user_id: int, cart_id: str, guest_items: list[dict]) -> None:
    """Гостевая корзина (Redis) → корзина аккаунта; количество из гостевой заменяет сохранённое для тех же товаров."""
    if not guest_items:
        return
    skus = [item["sku"] for item in guest_items]
    prod_map = {p.sku: p.id for p in (await db.execute(select(Product).where(Product.sku.in_(skus)))).scalars()}
    rows = [
        {"user_id": user_id, "product_id": prod_map[item["sku"]], "quantity": item["qty"]}
        for item in guest_items if item["sku"] in prod_map
    ]
    if rows:
        await db.execute(delete(UserCart).where(
            UserCart.user_id == user_id, UserCart.product_id.in_([r["product_id"] for r in rows])))
        await db.execute(insert(UserCart).values(rows))
        await db.commit()
    cart.redis_client.delete(cart._redis_key(cart_id))


def _reset_session(request: Request) -> None:
    """Новая сессия при входе (защита от session fixation), но гостевой диалог чата сохраняем —
    services/chat.find_thread привяжет его к аккаунту."""
    chat_token = request.session.get("chat_token")
    request.session.clear()
    if chat_token:
        request.session["chat_token"] = chat_token


async def _login_user(request: Request, db: AsyncSession, user: User) -> None:
    """Вход: новая сессия (защита от session fixation), гостевая корзина переезжает в аккаунт."""
    cart_id = cart.get_cart_id(request)
    guest_items = await cart.get_cart_items(request, db)
    _reset_session(request)
    request.session["user_id"] = user.id
    request.session["pwv"] = accounts.session_fingerprint(user)
    await _merge_guest_cart(db, user.id, cart_id, guest_items)


def _safe_next(value: str | None, default: str) -> str:
    """Куда вернуть после входа: только относительный путь этого сайта (защита от open redirect)."""
    if value and value.startswith("/") and not value.startswith("//") and "\\" not in value:
        return value
    return default

@router.post("/login")
async def login(
    request: Request,
    username: str = Form(...),
    password: str = Form(...),
    next: str = Form(""),
    db: AsyncSession = Depends(get_db),
):
    """Обрабатывает POST-форму входа. При успехе — редирект на главную."""
    username = username.strip()
    # Не больше 10 неудачных попыток за 15 минут на логин и на IP (задержка в 1 с не спасает от параллельного перебора)
    ip = request.client.host if request.client else "?"
    keys = [f"login_fail:u:{username.lower()}", f"login_fail:ip:{ip}"]
    if any(int(cart.redis_client.get(k) or 0) >= LOGIN_ATTEMPTS for k in keys):
        return templates.TemplateResponse(
            request=request,
            name="login.html",
            context={"error": "Слишком много попыток входа. Подождите 15 минут или войдите через Telegram.",
                     "next": next, "username": username},
            status_code=429,
        )
    result = await db.execute(select(User).where(User.username == username))
    user = result.scalar_one_or_none()

    if not user or len(password) > 128 or not verify_password(password, user.password_hash):
        for k in keys:
            cart.redis_client.incr(k)
            cart.redis_client.expire(k, 900)
        await asyncio.sleep(1) # Защита от брутфорса
        return templates.TemplateResponse(
            request=request,
            name="login.html",
            context={"error": "Неверное имя пользователя или пароль", "next": next, "username": username},
        )

    await _login_user(request, db, user)
    return RedirectResponse(url=_safe_next(next, "/catalog"), status_code=303)

@router.post("/register")
async def register(
    request: Request,
    username: str = Form(...),
    password: str = Form(...),
    next: str = Form(""),
    phone: str = Form(""),
    consent: str = Form(""),
    db: AsyncSession = Depends(get_db),
):
    """
    Регистрирует нового пользователя. Имя пользователя должно быть уникальным.
    После успешной регистрации сразу входим (пишем user_id в сессию).
    """
    username = username.strip()
    phone = phone.strip()[:50]
    if not consent:
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Нужно согласие на обработку персональных данных", "next": next, "username": username, "phone": phone},
        )
    if not (3 <= len(username) <= 50):
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Имя пользователя должно содержать от 3 до 50 символов", "next": next, "username": username},
        )
    if re.fullmatch(r"tg\d+(_.*)?", username, re.IGNORECASE):
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Имена вида «tg123» заняты под аккаунты из Telegram — выберите другое", "next": next, "username": username},
        )

    # Проверяем, не занято ли имя (без учёта регистра: «Admin» и «admin» — одно имя)
    result = await db.execute(select(User).where(func.lower(User.username) == username.lower()))
    if result.scalar_one_or_none():
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Пользователь с таким именем уже существует", "next": next, "username": username},
        )

    # Валидация длины пароля
    if not (8 <= len(password) <= 128):
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Пароль должен содержать от 8 до 128 символов", "next": next, "username": username},
        )

    new_user = User(username=username, password_hash=get_password_hash(password), phone=phone or None)
    try:
        db.add(new_user)
        await db.commit()
        await db.refresh(new_user)
    except Exception:
        await db.rollback()
        return templates.TemplateResponse(
            request=request,
            name="register.html",
            context={"error": "Ошибка при создании аккаунта. Попробуйте снова.", "next": next, "username": username},
        )

    # Сразу авторизуем нового пользователя, гостевую корзину сохраняем
    await _login_user(request, db, new_user)
    return RedirectResponse(url=_safe_next(next, "/auth/account"), status_code=303)

async def change_password(request: Request, old_password: str = Form(""), new_password: str = Form(...),
                          db: AsyncSession = Depends(get_db)):
    """Смена пароля. Аккаунт из Telegram-бота пароля не имеет — задаёт его без старого."""
    user = await get_current_user(request, db)
    if not user:
        return RedirectResponse(url="/auth/login", status_code=303)
    if accounts.has_password(user) and not verify_password(old_password, user.password_hash):
        return _account_back(error="Текущий пароль указан неверно")
    if not (8 <= len(new_password) <= 128):

    user.password_hash = get_password_hash(new_password)
    await db.commit()
    request.session["pwv"] = accounts.session_fingerprint(user)  # эта сессия остаётся, остальные выходят
    return _account_back(msg="Пароль сохранён. На других устройствах нужно войти заново.")

@router.get("/telegram", response_class=HTMLResponse)
def telegram_login_page(request: Request, next: str = ""):
    if not (telegram.enabled() and telegram.bot_link()):
        return RedirectResponse(url="/auth/login", status_code=303)
    # В токене — откуда пришёл запрос: бот покажет это и попросит подтвердить вход
    ua = (request.headers.get("user-agent") or "")[:120]
    ip = request.client.host if request.client else "?"
    token = telegram.new_token("login", "?" + json.dumps({"ip": ip, "ua": ua}, ensure_ascii=False))
    request.session["tg_login"] = token
    return templates.TemplateResponse(request=request, name="tg_login.html",
                                      context={"link": telegram.bot_link(f"login_{token}"), "next": _safe_next(next, "/catalog")})


@router.get("/telegram/poll")
async def telegram_login_poll(request: Request, db: AsyncSession = Depends(get_db)):
    token = request.session.get("tg_login")
    value = telegram.peek_token("login", token) if token else None
    if value is None:
        return JSONResponse({"ok": False, "expired": True})
    if not value.isdigit():
        return JSONResponse({"ok": False})
    telegram.take_token("login", token)
    user = (await db.execute(select(User).where(User.id == int(value)))).scalar_one_or_none()
    if not user:
        return JSONResponse({"ok": False, "expired": True})
    await _login_user(request, db, user)
    return JSONResponse({"ok": True})

PAGE_SIZE = 24
SUGGEST_LIMIT = 8


def _visible_filter():
    """Показываем все товары, кроме скрытых в админке (Product.is_active = 0); без цены — «Цена по запросу»."""
    hidden = exists().where(Product.sku == MobaPrice.sku, Product.is_active == 0)
    return [~hidden]


def _category_filter(category_ids: list[int]):
    return exists().where(Product.sku == MobaPrice.sku, Product.category_id.in_(category_ids))


# Примеры запросов под строкой поиска — показывают, что можно искать моделью и деталью
SEARCH_EXAMPLES = ["iPhone 11 дисплей", "realme c67 акб", "honor 10i дисплей в рамке", "iPhone 13 pro max шлейф", "A51 крышка"]


async def _stock_map(db: AsyncSession, skus: list[str], city_id: int | None) -> dict:
    """sku → SimpleNamespace(city=остаток в выбранном городе, total=всего) для карточек."""
    if not skus:
        return {}
    in_city = func.sum(case((Stock.city_id == city_id, Stock.quantity), else_=0)) if city_id else func.sum(0)
    rows = (
        await db.execute(
            select(Product.sku, in_city, func.sum(Stock.quantity))
            .join(Stock, Stock.product_id == Product.id)
            .where(Product.sku.in_(skus))
            .group_by(Product.sku)
        )
    ).all()
    return {sku: SimpleNamespace(city=int(c or 0), total=int(t or 0)) for sku, c, t in rows}


def _related_order(name: str) -> list:
    """Сначала товары для той же модели: совпадение латинских слов названия («iphone 11»)."""
    words = search.parse(name)["latin"][:2]  # обычно это бренд/линейка и номер модели
    if not words:
        return [MobaPrice.id]
    phrase = r"\m" + r"[\s/]+".join(words) + r"\M"
    return [case((MobaPrice.name.op("~*")(phrase), 0), else_=1), MobaPrice.id]


def _card_context(request: Request) -> dict:
    user = getattr(request.state, "user", None)
    return {"city": getattr(request.state, "city", None), "wholesale": bool(user and user.is_wholesale)}


def _parse_specs(raw: str | None) -> list[dict]:
    """specifications хранится JSON-строкой: список {key, value} или словарь."""
    if not raw:
        return []
    try:
        data = json.loads(raw)
    except (TypeError, ValueError):
        return []
    if isinstance(data, dict):
        return [{"key": k, "value": v} for k, v in data.items()]
    if isinstance(data, list):
        return [s for s in data if isinstance(s, dict) and "key" in s]
    return []


@router.get("", response_class=HTMLResponse)
async def catalog_list(
    request: Request,
    q: str | None = None,
    cat: str | None = None,
    page: int = 1,
    db: AsyncSession = Depends(get_db),
):
    """
    Каталог: разделы, фильтр по категории (?cat=slug, включая подкатегории), поиск (?q=) и пагинация.
    """
    page = max(page, 1)
    tree = await catalog_tree.get_tree(db)
    node = tree.by_slug.get(cat) if cat else None

    conditions = _visible_filter()
    if node:
        conditions.append(_category_filter(tree.descendant_ids(node)))
    if q and q.strip():
        conditions.extend(search.conditions(q))

    total = (await db.execute(select(func.count()).select_from(MobaPrice).where(*conditions))).scalar_one()
    total_pages = max(1, (total + PAGE_SIZE - 1) // PAGE_SIZE)
    page = min(page, total_pages)

    # Без поиска — сначала новые поступления
    order = search.order_by(q) if q and q.strip() else [search.availability().desc(), MobaPrice.id.desc()]
    query = (
        select(MobaPrice)
        .where(*conditions)
        .order_by(*order)
        .offset((page - 1) * PAGE_SIZE)
        .limit(PAGE_SIZE)
    )
    products = (await db.execute(query)).scalars().all()
    ctx = _card_context(request)
    stocks = await _stock_map(db, [p.sku for p in products], ctx["city"].id if ctx["city"] else None)

    return templates.TemplateResponse(
        request=request,
        name="catalog.html",
        context={
            "products": products,
            "total": total,
            "total_pages": total_pages,
            "page": page,
            "q": q,
            "cat": node.slug if node else None,
            "node": node,
            "path": tree.path(node) if node else [],
            "sections": tree.sections,
            "stocks": stocks,
            "examples": SEARCH_EXAMPLES,
            **ctx,
        },
    )

@router.get("/{sku}", response_class=HTMLResponse)
async def product_detail(
    request: Request,
    sku: str,
    db: AsyncSession = Depends(get_db),
):
    """
    Страница отдельного товара.
    """
    moba_price = (await db.execute(select(MobaPrice).where(MobaPrice.sku == sku))).scalars().first()
    product = (await db.execute(select(Product).where(Product.sku == sku))).scalar_one_or_none()

    if not moba_price or (product and not product.is_active):
        return templates.TemplateResponse(
            request=request,
            name="404.html",
            context={"title": "Товар не найден", "detail": f"Артикул {sku} отсутствует в каталоге."},
            status_code=404,
        )

    # Остатки по всем городам (Stock + City)
    stocks = []
    path = []
    if product:
        stock_stmt = (
            select(Stock, City)
            .join(City, City.id == Stock.city_id)
            .where(Stock.product_id == product.id)
            .order_by(City.id)
        )
        stocks = (await db.execute(stock_stmt)).all()
        tree = await catalog_tree.get_tree(db)
        node = tree.by_id.get(product.category_id)
        path = tree.path(node) if node else []

    # Похожие товары: та же категория, сначала с той же моделью в названии
    related = []
    if product:
        related = (
            await db.execute(
                select(MobaPrice)
                .where(*_visible_filter(), MobaPrice.sku != sku,
                       exists().where(Product.sku == MobaPrice.sku, Product.category_id == product.category_id))
                .order_by(*_related_order(moba_price.name))
                .limit(8)
            )
        ).scalars().all()
    ctx = _card_context(request)
    related_stocks = await _stock_map(db, [r.sku for r in related], ctx["city"].id if ctx["city"] else None)

    return templates.TemplateResponse(
        request=request,
        name="product.html",
        context={
            "product": product,
            "moba_price": moba_price,
            "specs": _parse_specs(moba_price.specifications),
            "stocks": stocks,
            "sku": sku,
            "path": path,
            "related": related,
            "related_stocks": related_stocks,
            **ctx,
        },
    )

@router.get("/state")
async def chat_state(request: Request, after: int = 0, read: int = 0, db: AsyncSession = Depends(get_db)):
    """Новые сообщения диалога (после id=after). read=1 — окно чата открыто, сбрасываем непрочитанные."""
    thread = await chat.find_thread(request, db)
    if not thread:
        return JSONResponse({"thread": None, "messages": [], "unread": 0})
    msgs = await chat.messages_after(db, thread.id, after)
    unread = thread.unread_client
    if read and unread:
        thread.unread_client = 0
        await db.commit()
        unread = 0
    return JSONResponse({
        "thread": {"id": thread.id, "status": thread.status.name, "has_contact": bool(thread.user_id or thread.phone)},
        "messages": [chat.serialize(m) for m in msgs],
        "unread": unread,
    })


@router.post("/send")
async def chat_send(request: Request, db: AsyncSession = Depends(get_db)):
    """Сообщение покупателя. Гость может передать name и phone, чтобы менеджер мог перезвонить."""
    try:
        data = await request.json()
    except ValueError:
        return JSONResponse({"ok": False, "error": "Некорректный запрос"}, status_code=400)
    ip = request.client.host if request.client else "?"
    if chat.rate_limited(("ip", ip), chat.IP_WINDOW, chat.IP_LIMIT):
        return JSONResponse({"ok": False, "error": "Слишком много сообщений — попробуйте позже"}, status_code=429)
    thread = await chat.get_or_create_thread(request, db)
    if chat.rate_limited(thread.id):
        return JSONResponse({"ok": False, "error": "Слишком часто — подождите несколько секунд"}, status_code=429)
    name = str(data.get("name") or "").strip()[:120]
    phone = str(data.get("phone") or "").strip()[:50]
    if name:
        thread.name = name
    if phone:
        thread.phone = phone
    page = str(data.get("page") or "")[:500]
    if page.startswith("/"):
        thread.last_page = page
    first_unread = not thread.unread_staff
    try:
        msg = await chat.add_message(db, thread, "client", str(data.get("text") or ""))
    except chat.ChatError as e:
        await db.rollback()
        return JSONResponse({"ok": False, "error": str(e)}, status_code=400)
    await db.commit()
    if first_unread:
        telegram.fire(telegram.notify_chat_message(thread.id, msg.text))
    return JSONResponse({"ok": True, "message": chat.serialize(msg)})

@router.get("", response_class=HTMLResponse)
async def checkout_page(
    request: Request,
    db: AsyncSession = Depends(get_db)
):
    """Страница оформления заказа."""
    # Проверка авторизации
    user_id = request.session.get("user_id")
    if not user_id:
        return RedirectResponse(url="/auth/login", status_code=303)

    items = await cart.get_cart_items(request, db)
    if not items:
        return RedirectResponse(url="/cart", status_code=303)

    cities = (await db.execute(select(City).where(City.is_active.is_(True)).order_by(City.id))).scalars().all()

    # Сколько позиций корзины можно собрать на складе каждого города — чтобы выбрать город без отказа
    need = {i["sku"]: i["qty"] for i in items}
    rows = (
        await db.execute(
            select(Product.sku, Stock.city_id, Stock.quantity)
            .join(Stock, Stock.product_id == Product.id)
            .where(Product.sku.in_(list(need)))
        )
    ).all()
    short = {c.id: len(need) for c in cities}
    for sku, city_id, qty in rows:
        if city_id in short and qty >= need[sku]:
            short[city_id] -= 1
    selected = getattr(request.state, "city", None)
    default_city = selected.id if selected and short.get(selected.id) == 0 else next(
        (c.id for c in cities if short[c.id] == 0), selected.id if selected else None)

    # Подставляем имя и телефон из последнего заказа
    last = (
        await db.execute(select(Order).where(Order.user_id == user_id).order_by(Order.id.desc()).limit(1))
    ).scalar_one_or_none()

    return templates.TemplateResponse(
        request=request,
        name="checkout.html",
        context={
            "items": items, "cities": cities, "short": short, "default_city": default_city,
            "prefill": {
                "name": last.customer_name if last else (request.state.user.full_name or "") if request.state.user else "",
                "phone": last.customer_phone if last else (request.state.user.phone or "") if request.state.user else "",
            },
            **cart.cart_totals(items),
        },
    )


@router.post("/process")
async def process_checkout(
    request: Request,
    customer_name: str = Form(...),
    customer_phone: str = Form(...),
    address: str = Form(None),
    city_id: int = Form(...),
    comment: str = Form(""),
    consent: str = Form(""),
    db: AsyncSession = Depends(get_db)
):
    """Обработка создания заказа."""
    user_id = request.session.get("user_id")
    if not user_id:
        return RedirectResponse(url="/auth/login", status_code=303)

    items = await cart.get_cart_items(request, db)
    if not items:
        return RedirectResponse(url="/cart", status_code=303)

    if not consent:
        return RedirectResponse(url="/cart?error=" + quote("Нужно согласие на обработку персональных данных"), status_code=303)
    customer_name, customer_phone = customer_name.strip()[:200], customer_phone.strip()[:50]
    if not customer_name or sum(ch.isdigit() for ch in customer_phone) < 5:
        return RedirectResponse(url="/cart?error=" + quote("Укажите имя и телефон для связи"), status_code=303)

    # Цена берётся с сервера (get_cart_items), а не из формы
    cart_data = [{"sku": item["sku"], "qty": item["qty"], "price": item["price"]} for item in items]
    user = request.state.user

    try:
        order = await order_service.create_order(
            db=db,
            user_id=user_id,
            city_id=city_id,
            cart_items=cart_data,
            customer_name=customer_name,
            customer_phone=customer_phone,
            address=(address or "").strip()[:500] or None,
            wholesale=bool(user and user.is_wholesale),
            comment=comment.strip()[:1000] or None,
        )
        # Очистка корзины — в той же транзакции, что и заказ
        await cart.clear_cart_async(request, db, commit=False)
        if user and not user.phone:
            user.phone = customer_phone  # подставим в следующий раз и покажем менеджеру
        await db.commit()
    except order_service.OrderError as e:
        return RedirectResponse(url=f"/cart?error={quote(str(e))}", status_code=303)

    telegram.fire(telegram.notify_new_order(order.id))

    return RedirectResponse(url=f"/checkout/success?order={order.id}", status_code=303)

    path.mkdir(parents=True, exist_ok=True)
    return path


def _safe_path(filename: str) -> Path | None:
    """Файл внутри каталога обмена (1С присылает и подпапки import_files/…); выход за пределы запрещён."""
    base = _upload_dir()
    target = (base / filename.replace("\\", "/").lstrip("/")).resolve()
    return target if target.is_relative_to(base) and target != base else None


def _check_basic_auth(request: Request) -> bool:
    header = request.headers.get("authorization", "")
    if not header.lower().startswith("basic ") or not settings.ONEC_EXCHANGE_PASSWORD:
        return False
    try:
        login, _, password = base64.b64decode(header[6:]).decode("utf-8").partition(":")
    except (ValueError, UnicodeDecodeError):
        return False
    return hmac.compare_digest(login, settings.ONEC_EXCHANGE_LOGIN) and hmac.compare_digest(
        password, settings.ONEC_EXCHANGE_PASSWORD
    )


def _session_ok(request: Request) -> bool:
    token = request.cookies.get(COOKIE_NAME, "")
    expires = _sessions.get(token)
    return bool(expires and expires > time.time())


@router.api_route("/1c_exchange", methods=["GET", "POST"])
@router.api_route("/1c_exchange.php", methods=["GET", "POST"], include_in_schema=False)
async def exchange(request: Request, type: str = "", mode: str = "", filename: str = "",
                   db: AsyncSession = Depends(get_db)) -> Response:
    if not settings.ONEC_EXCHANGE_ENABLED:
        return _text("failure\nОбмен с 1С выключен (ONEC_EXCHANGE_ENABLED=false)", 404)

    if mode == "checkauth":
        if not _check_basic_auth(request):
            logger.warning("1С: неверный логин/пароль обмена с %s", request.client.host if request.client else "?")
            return _text("failure\nНеверный логин или пароль", 401)
        token = onec_exchange.new_session_token()
        now = time.time()
        for t, exp in list(_sessions.items()):
            if exp < now:
                _sessions.pop(t, None)
        _sessions[token] = now + SESSION_TTL
        return _text(f"success\n{COOKIE_NAME}\n{token}")

    if not _session_ok(request):
        return _text("failure\nНе выполнена авторизация (checkauth)", 401)

    if mode == "init":
        if type == "catalog":
            shutil.rmtree(_upload_dir(), ignore_errors=True)  # новая выгрузка — старые файлы не нужны
        return _text(f"zip=no\nfile_limit={settings.ONEC_FILE_LIMIT}")

    if mode == "file":
        target = _safe_path(filename)
        if not target:
            return _text("failure\nНедопустимое имя файла", 400)
        target.parent.mkdir(parents=True, exist_ok=True)
        with open(target, "ab") as f:  # файл может приходить частями
            async for chunk in request.stream():
                f.write(chunk)
        return _text("success")

    if type == "catalog" and mode == "import":
        target = _safe_path(filename)
        if not target or not target.is_file():
            return _text(f"failure\nФайл {filename} не загружен", 400)
        try:
            data = target.read_bytes()
            name = os.path.basename(filename).lower()
            if name.startswith("offers") or name.startswith("prices") or name.startswith("rests"):
                result = await onec_exchange.import_offers(db, data)
            else:
                result = await onec_exchange.import_catalog(db, data)
            await db.commit()
        except Exception as e:
            await db.rollback()
            logger.exception("1С: ошибка импорта %s", filename)
            return _text(f"failure\n{e.__class__.__name__}: {e}")  # type здесь — параметр запроса
        logger.info("1С: импорт %s — %s", filename, result)
        return _text(f"success\n{result}")

    if type == "sale" and mode == "query":
        orders = await onec_exchange.pending_orders(db)
        _last_sent_orders[:] = [o.id for o in orders]
        return Response(onec_exchange.orders_xml(orders), media_type="application/xml; charset=utf-8")

    if type == "sale" and mode == "success":
        await onec_exchange.mark_exported(db, _last_sent_orders)
        await db.commit()
        logger.info("1С: подтверждено получение заказов %s", _last_sent_orders)
        _last_sent_orders.clear()
        return _text("success")

    return _text(f"failure\nНеизвестный режим type={type} mode={mode}", 400)

"""

к существующему аккаунту переносит в него всё, что было сделано в боте (заказы, корзину, переписку).

import hashlib

import json

import time
from urllib.parse import parse_qsl

from sqlalchemy import select, update, delete

from app.config import settings

# Пароль у аккаунта из бота не задан: вход через Telegram, пароль можно установить в личном кабинете
NO_PASSWORD = "!telegram"


class AccountError(Exception):
    pass


def session_fingerprint(user: User) -> str:
    """Отпечаток пароля для сессии: сменили пароль — старые сессии (cookie) перестают действовать."""
    return hashlib.sha256((user.password_hash or "").encode()).hexdigest()[:16]


def has_password(user: User) -> bool:
    return bool(user.password_hash) and user.password_hash != NO_PASSWORD


def is_auto_account(user: User) -> bool:
    """Аккаунт, созданный ботом, которым ещё не пользовались на сайте (без пароля)."""
    return not has_password(user) and user.username.startswith("tg")


async def telegram_user(db: AsyncSession, tg_id: int, full_name: str = "", tg_username: str = "") -> User:
    """Аккаунт по Telegram id; если его нет — создаётся (покупатель, розница)."""
    user = (await db.execute(select(User).where(User.telegram_id == tg_id))).scalar_one_or_none()
    if user:
        if full_name and not user.full_name:
            user.full_name = full_name[:200]
            await db.commit()
        return user
    username = f"tg{tg_id}"
    if (await db.execute(select(User.id).where(User.username == username))).first():
        username = f"tg{tg_id}_{secrets.token_hex(2)}"  # имя занято (до запрета «tg<цифры>» при регистрации)
    user = User(username=username, password_hash=NO_PASSWORD, telegram_id=tg_id,
                full_name=(full_name or tg_username or None) and (full_name or tg_username)[:200])
    db.add(user)
    await db.commit()
    await db.refresh(user)
    return user


async def link_telegram(db: AsyncSession, user: User, tg_id: int, full_name: str = "") -> None:
    """Привязать Telegram к аккаунту сайта. Если бот уже создал для этого Telegram отдельный аккаунт —
    заказы, корзина, переписка и заявки переезжают в аккаунт сайта, а автоаккаунт удаляется."""
    other = (await db.execute(select(User).where(User.telegram_id == tg_id))).scalar_one_or_none()
    if other and other.id != user.id:
        if not is_auto_account(other):
            raise AccountError("Этот Telegram уже привязан к другому аккаунту сайта")
        await _merge_into(db, other, user)
    user.telegram_id = tg_id
    if full_name and not user.full_name:
        user.full_name = full_name[:200]
    await db.commit()


async def _merge_into(db: AsyncSession, src: User, dst: User) -> None:
    await db.execute(update(Order).where(Order.user_id == src.id).values(user_id=dst.id))
    await db.execute(update(WholesaleApplication).where(WholesaleApplication.user_id == src.id).values(user_id=dst.id))
    await db.execute(update(InventoryTransaction).where(InventoryTransaction.user_id == src.id).values(user_id=dst.id))
    await db.execute(update(ChatMessage).where(ChatMessage.author_id == src.id).values(author_id=dst.id))
    # Корзина: позиции, которых нет у аккаунта сайта, переносим; совпадающие оставляем как есть
    have = set((await db.execute(select(UserCart.product_id).where(UserCart.user_id == dst.id))).scalars())
    for row in (await db.execute(select(UserCart).where(UserCart.user_id == src.id))).scalars().all():
        if row.product_id in have:
            await db.delete(row)
        else:
            row.user_id = dst.id
    # Переписка: в один диалог
    own = (await db.execute(select(ChatThread).where(ChatThread.user_id == dst.id)
                            .order_by(ChatThread.last_message_at.desc()).limit(1))).scalar_one_or_none()
    for thread in (await db.execute(select(ChatThread).where(ChatThread.user_id == src.id))).scalars().all():
        if own is None:
            thread.user_id = dst.id
            own = thread
        else:
            await db.execute(update(ChatMessage).where(ChatMessage.thread_id == thread.id).values(thread_id=own.id))
            own.unread_staff += thread.unread_staff
            own.unread_client += thread.unread_client
            await db.execute(delete(ChatThread).where(ChatThread.id == thread.id))
    await db.execute(update(ChatThread).where(ChatThread.assigned_to == src.id).values(assigned_to=dst.id))
    # Права не теряются: администратор (ADMIN_ID создаёт его автоаккаунт) остаётся администратором, оптовик — оптовиком
    if src.is_superuser:
        dst.role, dst.is_admin = UserRole.ADMIN, True
    elif src.role == UserRole.WHOLESALE and dst.role == UserRole.RETAIL:
        dst.role = UserRole.WHOLESALE
    dst.phone = dst.phone or src.phone
    src.telegram_id = None
    await db.flush()
    await db.delete(src)
    await db.flush()


def check_webapp_init_data(init_data: str, max_age: int = 3600) -> dict | None:
    """Подпись Telegram.WebApp.initData (Mini App). Возвращает объект user или None."""
    if not init_data or not settings.TELEGRAM_BOT_TOKEN:
        return None
    try:
        data = dict(parse_qsl(init_data, strict_parsing=True))
    except ValueError:
        return None
    received = data.pop("hash", "")
    check = "\n".join(f"{k}={v}" for k, v in sorted(data.items()))
    secret = hmac.new(b"WebAppData", settings.TELEGRAM_BOT_TOKEN.encode(), hashlib.sha256).digest()
    if not hmac.compare_digest(hmac.new(secret, check.encode(), hashlib.sha256).hexdigest(), received):
        return None
    try:
        if time.time() - int(data.get("auth_date", "0")) > max_age:
            return None
        user = json.loads(data.get("user", "null"))
    except ValueError:
        return None
    return user if isinstance(user, dict) and isinstance(user.get("id"), int) else None

def _sold_subquery(since: datetime | None = None):

    qty = func.coalesce(st.c.qty, 0)
    row = (
        await db.execute(
            select(
                func.count(Product.id).label("total_parts"),
                func.coalesce(func.sum(qty), 0).label("total_stock"),
                func.coalesce(func.sum(qty * func.coalesce(MobaPrice.our_base, 0)), 0).label("retail_value"),
                func.coalesce(func.sum(qty * func.coalesce(Product.purchase_price, 0)), 0).label("cost_value"),
                func.coalesce(func.sum(case((and_(qty <= Product.critical_stock_threshold, stocked()), 1), else_=0)), 0).label("low_stock"),
            )
            .select_from(Product)
            .outerjoin(st, st.c.product_id == Product.id)
            .outerjoin(MobaPrice, MobaPrice.sku == Product.sku)
            .where(Product.is_active == 1)
        )
    ).one()
    retail, cost = Decimal(row.retail_value), Decimal(row.cost_value)
    margin = retail - cost
    return {
        "total_parts": row.total_parts,
        "total_stock": int(row.total_stock),
        "retail_value": retail,
        "cost_value": cost,
        "margin_value": margin,
        "margin_pct": float(margin / retail * 100) if retail > 0 else 0.0,
        "low_stock": int(row.low_stock),
        "total_users": (await db.execute(select(func.count(User.id)))).scalar_one(),
        "pending_orders": (
            await db.execute(select(func.count(Order.id)).where(Order.status == OrderStatus.PENDING))
        ).scalar_one(),
        "pending_wholesale": (
            await db.execute(
                select(func.count(WholesaleApplication.id)).where(
                    WholesaleApplication.status == ApplicationStatus.PENDING
                )
            )
        ).scalar_one(),
    }


# ───────────────────────── KPI продаж ─────────────────────────


async def sales_kpi(db: AsyncSession) -> dict:
    now = datetime.now()
    today_start = now.replace(hour=0, minute=0, second=0, microsecond=0)

    async def period(since: datetime | None) -> dict:
        q = select(func.count(Order.id), func.coalesce(func.sum(Order.total_amount), 0)).where(
            Order.status != OrderStatus.CANCELLED
        )
        if since is not None:
            q = q.where(Order.created_at >= since)
        count, revenue = (await db.execute(q)).one()
        revenue = Decimal(revenue)
        return {"count": count, "revenue": revenue, "avg_check": revenue / count if count else ZERO}

    status_rows = (await db.execute(select(Order.status, func.count(Order.id)).group_by(Order.status))).all()
    return {
        "today": await period(today_start),
        "week": await period(now - timedelta(days=7)),
        "month": await period(now - timedelta(days=30)),
        "all": await period(None),
        "status_dist": {s.name: c for s, c in status_rows},
    }

# ───────────────────────── ABC-анализ ─────────────────────────


async def abc_analysis(db: AsyncSession) -> dict:
    """
    A — товары, дающие первые 80% накопленной выручки, B — следующие 15%,
    C — последние 5% и все активные товары без продаж.
    """
    sold = _sold_subquery()
    st = _stock_subquery()
    rows = (
        await db.execute(
            select(Product.id, Product.sku, Product.name, func.coalesce(st.c.qty, 0).label("stock_qty"),
                   sold.c.sold_qty, sold.c.revenue)
            .join(sold, sold.c.product_id == Product.id)
            .outerjoin(st, st.c.product_id == Product.id)
            .order_by(desc(sold.c.revenue))
        )
    ).all()
    total_revenue = sum((Decimal(r.revenue) for r in rows), ZERO)
    total_active = (await db.execute(select(func.count(Product.id)).where(Product.is_active == 1))).scalar_one()

    groups = {"A": [], "B": [], "C": []}
    cum = ZERO
    for r in rows:
        rev = Decimal(r.revenue)
        cum += rev
        pct = float(cum / total_revenue * 100) if total_revenue else 100.0
        g = "A" if pct <= 80 else "B" if pct <= 95 else "C"
        groups[g].append({
            "id": r.id, "sku": r.sku, "name": r.name, "stock_qty": int(r.stock_qty),
            "sold_qty": int(r.sold_qty), "revenue": rev,
            "share": float(rev / total_revenue * 100) if total_revenue else 0.0,
            "cumulative": pct, "group": g,
        })

    unsold = max(0, total_active - len(rows))
    # Неликвид: товары без продаж, но с наибольшим остатком — кандидаты на акции
    sold_ids = select(OrderItem.product_id).join(Order).where(Order.status != OrderStatus.CANCELLED)
    dead_stock = (
        await db.execute(
            select(Product.id, Product.sku, Product.name, st.c.qty)
            .join(st, st.c.product_id == Product.id)
            .where(Product.is_active == 1, st.c.qty > 0, Product.id.not_in(sold_ids))
            .order_by(desc(st.c.qty))
            .limit(10)
        )
    ).all()

    revenue = {g: sum((i["revenue"] for i in items), ZERO) for g, items in groups.items()}
    return {
        "total_revenue": total_revenue,
        "groups": groups,
        "counts": {"A": len(groups["A"]), "B": len(groups["B"]), "C": len(groups["C"]) + unsold},
        "unsold": unsold,
        "revenue": revenue,
        "pct": {g: float(v / total_revenue * 100) if total_revenue else 0.0 for g, v in revenue.items()},
        "dead_stock": dead_stock,
    }

URGENCY_LABELS = {
    "EMPTY": "Закончился",
    "HIGH": "< 7 дней",
    "MEDIUM": "< 14 дней",
    "OK": "В норме",
}


async def procurement_forecast(db: AsyncSession, days_window: int = 30, only_urgent: bool = True) -> list[dict]:
    """
    Темп расхода (продажи за days_window дней), запас в днях и рекомендуемый дозаказ
    до 30-дневного запаса. Логика повторяет db.get_procurement_forecast бота.
    """
    sold = _sold_subquery(datetime.now() - timedelta(days=days_window))
    st = _stock_subquery()
    rows = (
        await db.execute(
            select(Product.id, Product.sku, Product.name, Product.critical_stock_threshold,
                   Product.purchase_price, Supplier.name.label("supplier"),
                   func.coalesce(st.c.qty, 0).label("stock_qty"),
                   func.coalesce(sold.c.sold_qty, 0).label("sold"))
            .outerjoin(st, st.c.product_id == Product.id)
            .outerjoin(sold, sold.c.product_id == Product.id)
            .outerjoin(Supplier, Supplier.id == Product.supplier_id)
            .where(Product.is_active == 1)
        )
    ).all()

    result = []
    for r in rows:
        stock, threshold = int(r.stock_qty), r.critical_stock_threshold
        burn = int(r.sold) / days_window
        if burn > 0:
            days_left = round(stock / burn, 1)
            recommended = max(max(0, int(burn * 30) - stock), threshold * 2 - stock if stock <= threshold else 0)
        else:
            days_left = None if stock > 0 else 0.0
            recommended = threshold * 2 if stock == 0 else 0

        if stock == 0:
            urgency = "EMPTY"
        elif (days_left is not None and days_left <= 7) or stock <= threshold:
            urgency = "HIGH"
        elif days_left is not None and days_left <= 14:
            urgency = "MEDIUM"
        else:
            urgency = "OK"
        if only_urgent and urgency == "OK":
            continue
        result.append({
            "id": r.id, "sku": r.sku, "name": r.name, "stock_qty": stock, "threshold": threshold,
            "daily_burn": round(burn, 2), "days_left": days_left, "recommended": recommended,
            "supplier": r.supplier, "purchase_price": r.purchase_price,
            "urgency": urgency, "urgency_label": URGENCY_LABELS[urgency],
        })

    weight = {"EMPTY": 0, "HIGH": 1, "MEDIUM": 2, "OK": 3}
    result.sort(key=lambda x: (weight[x["urgency"]], x["days_left"] if x["days_left"] is not None else 9999))
    return result

async def movement_report(db: AsyncSession, days: int = 30) -> list:

async def finance_summary(db: AsyncSession) -> dict:
    completed = (
        await db.execute(
            select(
                func.coalesce(func.sum(Order.total_amount), 0),
                func.coalesce(func.sum(Order.profit), 0),
                func.count(Order.id),

        await db.execute(
            select(func.avg((MobaPrice.our_base - Product.purchase_price) * 100 / MobaPrice.our_base))
            .select_from(Product)
            .join(MobaPrice, MobaPrice.sku == Product.sku)
            .where(Product.is_active == 1, Product.purchase_price.isnot(None), MobaPrice.our_base > 0)
        )
    ).scalar()
    avg_cost = (await db.execute(select(func.avg(Product.purchase_price)))).scalar()
    revenue, profit = Decimal(completed[0]), Decimal(completed[1])
    return {
        "revenue": revenue,
        "profit": profit,
        "order_count": completed[2],
        "profit_pct": float(profit / revenue * 100) if revenue > 0 else 0.0,
        "avg_margin": float(avg_margin or 0),
        "avg_cost": Decimal(avg_cost or 0).quantize(Decimal("0.01")),
    }

"""
Гибридная корзина: Redis для гостей, БД (user_cart) для авторизованных пользователей.

Функции, меняющие корзину в БД, делают commit сами — роутеры получают сессию
через get_db(), которая коммит не выполняет.

from decimal import Decimal, ROUND_HALF_UP
from uuid import uuid4

from app.config import REDIS_URL
from app.models import MobaPrice, UserCart, Product

# Клиент Redis — синхронный, декодируем значения в строки
redis_client = redis.from_url(REDIS_URL, decode_responses=True)

CART_TTL = 60 * 60 * 24 * 30  # 30 дней
MAX_QTY = 999  # столько же разрешает поле количества на сайте


def get_cart_id(request) -> str:
    """Возвращает cart_id из сессии, создавая при отсутствии."""
    if "cart_id" not in request.session:
        request.session["cart_id"] = str(uuid4())
    return request.session["cart_id"]


def _redis_key(cart_id: str) -> str:
    return f"cart:{cart_id}"


def _is_wholesale(request) -> bool:
    user = getattr(request.state, "user", None)
    return bool(user and user.is_wholesale)


def _make_item(p: MobaPrice, qty: int, wholesale: bool) -> dict:
    prices = buyer_prices(p, wholesale)  # оптовику — опт (и картой — опт с наценкой карты)
    price, card = prices["price"] or Decimal(0), prices["card"] or Decimal(0)
    return {
        "sku": p.sku,
        "name": p.name,
        "qty": qty,
        "price": price,
        "sum": price * qty,
        "price_card": card,
        "sum_card": card * qty,
        "wholesale": prices["wholesale"],
    }


async def user_cart_items(db: AsyncSession, user_id: int, wholesale: bool) -> list[dict]:
    """Корзина аккаунта (сайт после входа и Telegram-бот — одна и та же корзина)."""
    stmt = (
        select(UserCart, MobaPrice)
        .join(Product, Product.id == UserCart.product_id)
        .join(MobaPrice, Product.sku == MobaPrice.sku)
        .where(UserCart.user_id == user_id, Product.is_active == 1, MobaPrice.our_base.isnot(None))
        .order_by(UserCart.id)
    )
    rows = (await db.execute(stmt)).all()
    return [_make_item(mp, ci.quantity, wholesale) for ci, mp in rows]


async def get_cart_items(request, db: AsyncSession) -> list[dict]:
    """Список позиций корзины: из БД для авторизованного пользователя, иначе из Redis."""
    wholesale = _is_wholesale(request)
    user_id = request.session.get("user_id")

    if user_id:
        return await user_cart_items(db, user_id, wholesale)

    raw = redis_client.hgetall(_redis_key(get_cart_id(request)))
    if not raw:
        return []
    result = await db.execute(
        select(MobaPrice)
        .join(Product, Product.sku == MobaPrice.sku)
        .where(MobaPrice.sku.in_(list(raw.keys())), Product.is_active == 1, MobaPrice.our_base.isnot(None))
    )
    by_sku = {p.sku: p for p in result.scalars().all()}

    items = []
    for sku, qty_str in raw.items():
        qty = int(qty_str)
        p = by_sku.get(sku)
        if p and qty > 0:
            items.append(_make_item(p, qty, wholesale))
    return items


async def _get_product(db: AsyncSession, sku: str) -> Product | None:
    """Товар, который можно купить: есть в базе, не скрыт и с ценой (не «Цена по запросу»)."""
    return (
        await db.execute(
            select(Product)
            .join(MobaPrice, MobaPrice.sku == Product.sku)
            .where(Product.sku == sku, Product.is_active == 1, MobaPrice.our_base.isnot(None))
        )
    ).scalar_one_or_none()


async def _get_cart_row(db: AsyncSession, user_id: int, product_id: int) -> UserCart | None:
    stmt = select(UserCart).where(UserCart.user_id == user_id, UserCart.product_id == product_id)
    return (await db.execute(stmt)).scalar_one_or_none()


async def user_add(db: AsyncSession, user_id: int, sku: str, qty: int) -> int:
    """Добавляет товар в корзину аккаунта; возвращает новое количество (0 — купить нельзя)."""
    if qty <= 0:
        return 0
    product = await _get_product(db, sku)
    if not product:
        return 0
    row = await _get_cart_row(db, user_id, product.id)
    qty = min(qty, MAX_QTY)
    if row:
        row.quantity = min(row.quantity + qty, MAX_QTY)
        new_qty = row.quantity
    else:
        db.add(UserCart(user_id=user_id, product_id=product.id, quantity=qty))
        new_qty = qty
    await db.commit()
    return new_qty


async def user_set(db: AsyncSession, user_id: int, sku: str, qty: int) -> int:

"""
Дерево каталога: 6 разделов (корневые категории с иконкой) → группы → подгруппы.
Количество товаров считается по видимым товарам витрины и кэшируется в памяти процесса.
"""
import time
from dataclasses import dataclass, field

from sqlalchemy import select, func
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Category, MobaPrice, Product

CACHE_TTL = 300  # секунд


@dataclass
class Node:
    id: int
    name: str
    slug: str
    icon: str | None
    position: int
    parent_id: int | None
    own_count: int = 0
    count: int = 0  # вместе с потомками
    children: list["Node"] = field(default_factory=list)


@dataclass
class Tree:
    sections: list[Node]
    by_slug: dict[str, Node]
    by_id: dict[int, Node]

    def descendant_ids(self, node: Node) -> list[int]:
        ids, stack = [], [node]
        while stack:
            n = stack.pop()
            ids.append(n.id)
            stack.extend(n.children)
        return ids

    def path(self, node: Node) -> list[Node]:
        out = []
        while node:
            out.append(node)
            node = self.by_id.get(node.parent_id) if node.parent_id else None
        return list(reversed(out))


_cache: tuple[float, Tree] | None = None


def invalidate() -> None:
    global _cache
    _cache = None


async def get_tree(db: AsyncSession) -> Tree:
    global _cache
    if _cache and time.monotonic() - _cache[0] < CACHE_TTL:
        return _cache[1]

    cats = (await db.execute(select(Category))).scalars().all()
    counts = dict(
        (
            await db.execute(
                select(Product.category_id, func.count(Product.id))
                .join(MobaPrice, MobaPrice.sku == Product.sku)
                .where(Product.is_active == 1)
                .group_by(Product.category_id)
            )
        ).all()
    )

    by_id = {c.id: Node(c.id, c.name, c.slug, c.icon, c.position, c.parent_id, counts.get(c.id, 0)) for c in cats}
    for n in by_id.values():
        if n.parent_id in by_id:
            by_id[n.parent_id].children.append(n)

    def total(n: Node) -> int:
        n.count = n.own_count + sum(total(c) for c in n.children)
        n.children = sorted((c for c in n.children if c.count), key=lambda c: (c.position, c.name))
        return n.count

    roots = [n for n in by_id.values() if n.parent_id is None]
    for r in roots:
        total(r)
    # Разделы — корневые категории с иконкой; пустые технические корни moba не показываем
    sections = sorted((r for r in roots if r.icon), key=lambda r: r.position)
    tree = Tree(sections, {n.slug: n for n in by_id.values()}, by_id)
    _cache = (time.monotonic(), tree)
    return tree

def rate_limited(key, window: int = RATE_WINDOW, limit: int = RATE_LIMIT) -> bool:
    now = time.monotonic()
    q = _recent[key]
    while q and now - q[0] > window:
        q.popleft()
    if len(q) >= limit:
        return True
    q.append(now)
    if len(_recent) > 10_000:  # чистим ключи без свежих сообщений, чтобы словарь не рос бесконечно
        for k in [k for k, v in _recent.items() if not v or now - v[-1] > IP_WINDOW]:

    return False

def extract_products_from_html(html: str) -> list[dict]:
    """
    Ищет в inline-скриптах JSON-массив товаров и возвращает его.

    Товары лежат в скрипте, который начинается с `[{"id":` и содержит поля
    article, name, price. Ищем по этим признакам, чтобы не путать с другими массивами.
    """
    scripts = re.findall(r"<script(?![^>]*\bsrc=)[^>]*>(.*?)</script>", html, re.DOTALL)

    for script in scripts:
        # Ищем массив, начинающийся с [{ "id": ..., "link": ... или "article": ...
        match = re.search(
            r'\[\s*\{\s*"id"\s*:\s*\d+\s*,\s*"link"\s*:',
            script,
        )
        if not match:
            continue

        start = match.start()
        # Находим соответствующий закрывающий ]
        depth = 0
        end = start
        for i in range(start, len(script)):
            if script[i] == "[":
                depth += 1
            elif script[i] == "]":
                depth -= 1
                if depth == 0:
                    end = i + 1
                    break

        array_str = script[start:end]
        try:
            products = json.loads(array_str)
            if products and "article" in products[0] and "price" in products[0]:
                return products
        except json.JSONDecodeError:
            continue

    return []

async def parse_category(page, category_path: str) -> list[dict]:
    """
    Парсит одну категорию целиком, обходя все страницы пагинации.
    Возвращает список товаров с добавленным полем url.
    """
    all_products: list[dict] = []
    seen_articles: set[str] = set()
    page_num = 1

    while True:
        if page_num == 1:
            url = f"{MOBA_BASE}{category_path}"
        else:
            url = f"{MOBA_BASE}{category_path}?page={page_num}"

        print(f"  → {url}")

        try:
            html = await fetch_page_html(page, url)
        except Exception as e:
            print(f"     ошибка загрузки: {e}")
            break

        products = extract_products_from_html(html)

        if not products:
            print(f"     товаров нет — стоп")
            break

        # Добавляем url к каждому товару
        for p in products:
            p["_parsed_url"] = url

        # Дедупликация внутри категории (одна и та же карточка может попасть на несколько страниц)
        new_products = []
        for p in products:
            article = p.get("article") or ""
            if article and article in seen_articles:
                continue
            if article:
                seen_articles.add(article)
            new_products.append(p)

        if not new_products:
            print(f"     все товары уже видели — стоп")
            break

        print(f"     найдено товаров: {len(new_products)}")
        all_products.extend(new_products)

        page_num += 1

        # Защита от бесконечного цикла
        if page_num > 100:
            print("     превышен лимит страниц (100)")
            break

        # Небольшая пауза между запросами, чтобы не злить moba
        await asyncio.sleep(3)

    return all_products


def save_products(db: Session, products: list[dict]) -> int:
    """
    Сохраняет товары в БД. Если запись с таким sku уже есть — обновляет её.
    Возвращает количество обработанных записей.
    """
    saved = 0
    for p in products:
        article = p.get("article")
        name = p.get("name")
        price = p.get("price")
        discount = p.get("discountPrice")
        url = p.get("_parsed_url")
        image = p.get("image")
        link = p.get("link")

        # Характеристики и теги — сохраняем как JSON-строки
        specifications = p.get("specifications")
        tags = p.get("tags")
        specs_json = json.dumps(specifications, ensure_ascii=False) if specifications else None
        tags_json = json.dumps(tags, ensure_ascii=False) if tags else None

        if not article or price is None:
            continue

        existing = db.query(MobaPrice).filter(MobaPrice.sku == article).first()

        if existing:
            existing.name = name
            existing.price = Decimal(str(price))
            existing.discount_price = Decimal(str(discount)) if discount else None
            existing.url = url
            existing.image = image
            existing.link = link
            existing.specifications = specs_json
            existing.tags = tags_json
        else:
            record = MobaPrice(
                sku=article,
                name=name,
                price=Decimal(str(price)),
                discount_price=Decimal(str(discount)) if discount else None,
                url=url,
                image=image,
                link=link,
                specifications=specs_json,
                tags=tags_json,
            )
            db.add(record)

        saved += 1

    db.commit()
    return saved

        return (f"создано {self.created}, обновлено {self.updated}, цен {self.prices}, остатков {self.stocks}"
                + (f", пропущено {len(self.skipped)}" if self.skipped else ""))


# ───────────────────────── XML-утилиты ─────────────────────────


def _strip_ns(root: ET.Element) -> ET.Element:
    """CommerceML бывает с пространством имён urn:1C.ru:commerceml_2 и без — приводим к именам без него."""
    for el in root.iter():
        if isinstance(el.tag, str) and "}" in el.tag:
            el.tag = el.tag.split("}", 1)[1]
    return root


def _text(el: ET.Element | None, path: str, default: str = "") -> str:
    found = el.find(path) if el is not None else None
    return (found.text or "").strip() if found is not None and found.text else default


def _decimal(value: str) -> Decimal | None:
    try:
        return Decimal(value.replace(",", ".").replace("\xa0", "").replace(" ", "")) if value else None
    except InvalidOperation:
        return None


def _product_id(offer_id: str) -> str:
    """Ид предложения бывает «ИдТовара#ИдХарактеристики» — остатки и цены относим к товару."""
    return offer_id.split("#", 1)[0]

async def import_catalog(db: AsyncSession, xml_bytes: bytes) -> ImportResult:
    """import.xml: создаёт/обновляет товары. Коммит делает вызывающий код."""
    root = _strip_ns(safe_fromstring(xml_bytes))
    result = ImportResult()
    fallback_cat = None
    group_to_cat = {
        c.onec_id: c.id for c in (await db.execute(select(Category).where(Category.onec_id.isnot(None)))).scalars()
    }

    for item in root.iter("Товар"):
        onec_id = _text(item, "Ид")
        name = _text(item, "Наименование")
        sku = _text(item, "Артикул") or onec_id
        if not onec_id or not name:
            result.skipped.append(onec_id or "?")
            continue
        if item.get("Статус") == "Удален" or _text(item, "ПометкаУдаления").lower() == "true":
            p = await _find_product(db, onec_id, sku)
            if p:
                p.is_active = 0
                result.updated += 1
            continue

        group_ids = [g.text.strip() for g in item.findall("Группы/Ид") if g.text]
        category_id = next((group_to_cat[g] for g in group_ids if g in group_to_cat), None)

        product = await _find_product(db, onec_id, sku)
        if product is None:
            if category_id is None:
                fallback_cat = fallback_cat or await _new_items_category(db)
                category_id = fallback_cat.id
            slug = re.sub(r"[^a-z0-9]+", "-", sku.lower()).strip("-") or onec_id
            product = Product(sku=sku, name=name, slug=slug, category_id=category_id, is_active=1, onec_id=onec_id)
            db.add(product)
            result.created += 1
        else:
            product.onec_id = onec_id
            product.name = name
            if category_id:
                product.category_id = category_id
            result.updated += 1
        description = _text(item, "Описание")
        if description:
            product.description = description

        mp = (await db.execute(select(MobaPrice).where(MobaPrice.sku == product.sku))).scalars().first()
        if mp:
            mp.name = name
        else:
            # Витрина читает moba_prices; цена появится после offers.xml
            db.add(MobaPrice(sku=product.sku, name=name, price=0))
        await db.flush()

    catalog_tree.invalidate()
    return result


async def import_offers(db: AsyncSession, xml_bytes: bytes) -> ImportResult:
    """offers.xml: цены и остатки по складам. Коммит делает вызывающий код."""
    root = _strip_ns(safe_fromstring(xml_bytes))
    result = ImportResult()

    price_types = {_text(t, "Ид"): _text(t, "Наименование") for t in root.iter("ТипЦены")}
    base_type = next((i for i, n in price_types.items() if n == settings.ONEC_PRICE_TYPE_BASE), None)
    opt_type = next((i for i, n in price_types.items() if n == settings.ONEC_PRICE_TYPE_OPT), None)
    if price_types and base_type is None:
        logger.warning("1С: вид цены «%s» не найден среди %s", settings.ONEC_PRICE_TYPE_BASE, list(price_types.values()))
    coef_card = Decimal(await get_config_value(db, "COEF_CARD", "1.1"))

    cities = (await db.execute(select(City))).scalars().all()
    warehouses = {_text(w, "Ид"): _text(w, "Наименование") for w in root.iter("Склад") if _text(w, "Ид")}
    city_by_wh: dict[str, int] = {}
    for wh_id, wh_name in warehouses.items():
        city = next((c for c in cities if c.onec_warehouse_id == wh_id), None) or next(
            (c for c in cities if c.name.lower() in wh_name.lower()), None
        )
        if city:
            city_by_wh[wh_id] = city.id
    for c in cities:
        if c.onec_warehouse_id:
            city_by_wh.setdefault(c.onec_warehouse_id, c.id)

    for offer in root.iter("Предложение"):
        onec_id = _product_id(_text(offer, "Ид"))
        product = await _find_product(db, onec_id, _text(offer, "Артикул"))
        if not product:
            result.skipped.append(onec_id)
            continue

        prices = {}
        for price in offer.findall("Цены/Цена"):
            value = _decimal(_text(price, "ЦенаЗаЕдиницу"))
            if value is not None:
                prices[_text(price, "ИдТипаЦены")] = value
        base = prices.get(base_type) if base_type else (next(iter(prices.values()), None))
        if base is not None:
            opt = prices.get(opt_type) if opt_type else None
            mp = (await db.execute(select(MobaPrice).where(MobaPrice.sku == product.sku))).scalars().first()
            if mp is None:
                mp = MobaPrice(sku=product.sku, name=product.name, price=base)
                db.add(mp)
            mp.price = base
            mp.our_base = round_money(base)
            mp.our_opt = round_money(opt) if opt is not None else round_money(base)
            mp.our_card = round_money(base * coef_card)
            product.price_base, product.price_opt, product.price_card = mp.our_base, mp.our_opt, mp.our_card
            result.prices += 1

        # Остатки: <Склад ИдСклада=".." КоличествоНаСкладе=".."/> или <Остатки><Остаток><Склад><Ид/><Количество/>
        per_city: dict[int, int] = {}
        for wh in offer.findall("Склад"):
            city_id = city_by_wh.get(wh.get("ИдСклада", ""))
            qty = _decimal(wh.get("КоличествоНаСкладе", ""))
            if city_id and qty is not None:
                per_city[city_id] = per_city.get(city_id, 0) + int(qty)
        for rest in offer.findall("Остатки/Остаток/Склад"):
            city_id = city_by_wh.get(_text(rest, "Ид"))
            qty = _decimal(_text(rest, "Количество"))
            if city_id and qty is not None:
                per_city[city_id] = per_city.get(city_id, 0) + int(qty)
        if not per_city and settings.ONEC_DEFAULT_CITY_ID:
            qty = _decimal(_text(offer, "Количество"))
            if qty is not None:
                per_city[settings.ONEC_DEFAULT_CITY_ID] = int(qty)

        for city_id, qty in per_city.items():
            qty = max(qty, 0)
            stock = (
                await db.execute(select(Stock).where(Stock.product_id == product.id, Stock.city_id == city_id))
            ).scalar_one_or_none()
            old = stock.quantity if stock else 0
            if stock:
                stock.quantity = qty
            else:
                db.add(Stock(product_id=product.id, city_id=city_id, quantity=qty))
            if qty != old:
                db.add(InventoryTransaction(
                    product_id=product.id, type=TransactionType.ADJUSTMENT, quantity=qty - old,
                    to_city_id=city_id if qty > old else None, from_city_id=city_id if qty < old else None,
                    notes="Остаток из 1С",
                ))
            result.stocks += 1
        await db.flush()

    catalog_tree.invalidate()
    return result

"""
Сервис управления заказами.
Отвечает за атомарное создание заказов и управление остатками на складе.
"""
import logging
from decimal import Decimal

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.models.city import City
from app.models.inventory_transaction import InventoryTransaction, TransactionType
from app.models.order import Order, OrderItem, OrderStatus, OrderType
from app.models.product import Product
from app.models.stock import Stock

logger = logging.getLogger(__name__)


class OrderError(Exception):
    """Бизнес-ошибка оформления/изменения заказа (текст можно показывать пользователю)."""


async def create_order(
    db: AsyncSession,
    user_id: int | None,
    city_id: int,
    cart_items: list[dict],  # [{sku, qty, price}]
    customer_name: str,
    customer_phone: str,
    address: str | None = None,
    wholesale: bool = False,
    comment: str | None = None,
    source: str = "site",
) -> Order:
    """
    Создаёт заказ и атомарно списывает остатки со склада выбранного города.

    Транзакция: строки остатков блокируются SELECT ... FOR UPDATE, при любой ошибке
    выполняется rollback и ни одно списание не сохраняется. Коммит делает вызывающий
    код (чтобы в ту же транзакцию попала очистка корзины).
    """
    try:
        city = await db.get(City, city_id)
        if not city or not city.is_active:
            raise OrderError("Выберите склад для получения заказа")
        if not cart_items:
            raise OrderError("Корзина пуста")
        total_amount = Decimal("0.00")
        profit = Decimal("0.00")
        order = Order(
            user_id=user_id,
            city_id=city_id,
            total_amount=0,
            status=OrderStatus.PENDING,
            order_type=OrderType.WHOLESALE if wholesale else OrderType.RETAIL,
            customer_name=customer_name,
            customer_phone=customer_phone,
            address=address,
            comment=comment,
            source=source,
        )
        db.add(order)

        for item in cart_items:
            sku, qty = item["sku"], int(item["qty"])
            product = (await db.execute(select(Product).where(Product.sku == sku))).scalar_one_or_none()
            if not product:
                raise OrderError(f"Товар {sku} не найден в базе")
            if not product.is_active:
                raise OrderError(f"Товар «{product.name}» снят с продажи — удалите его из корзины")
            if qty <= 0:
                raise OrderError("Некорректное количество товара")
            if Decimal(item["price"]) <= 0:
                raise OrderError(f"У товара «{product.name}» нет цены — уточните её у менеджера в чате")

            stock = (
                await db.execute(
                    select(Stock)
                    .where(Stock.product_id == product.id, Stock.city_id == city_id)
                    .with_for_update()
                )
            ).scalar_one_or_none()
            if not stock or stock.quantity < qty:
                available = stock.quantity if stock else 0
                raise OrderError(f"Недостаточно товара «{product.name}» в выбранном городе (доступно: {available} шт.)")

            stock.quantity -= qty
            price = Decimal(item["price"])
            total_amount += price * qty
            if product.purchase_price is not None:
                profit += (price - Decimal(product.purchase_price)) * qty

            order.items.append(OrderItem(product_id=product.id, quantity=qty, price_at_time=price))
            db.add(InventoryTransaction(
                product_id=product.id, user_id=user_id, type=TransactionType.SALE,
                quantity=-qty, from_city_id=city_id,
            ))

        order.total_amount = total_amount
        order.profit = profit
        await db.flush()  # получаем order.id
        return order
    except OrderError:
        await db.rollback()
        raise
    except Exception:
        await db.rollback()
        logger.exception("create_order failed")
        raise OrderError("Внутренняя ошибка при оформлении заказа")


async def change_order_status(db: AsyncSession, order_id: int, new_status: OrderStatus, staff_id: int | None) -> Order:
    """
    Меняет статус заказа с учётом склада:
    - переход в CANCELLED возвращает товар на склад города заказа;
    - выход из CANCELLED снова списывает товар (если его хватает).
    Коммит делает вызывающий код.
    """
    order = (
        await db.execute(select(Order).options(selectinload(Order.items)).where(Order.id == order_id))
    ).scalar_one_or_none()
    if not order:
        raise OrderError("Заказ не найден")
    old_status = order.status
    if old_status == new_status:
        return order

    if new_status == OrderStatus.CANCELLED or old_status == OrderStatus.CANCELLED:
        sign = 1 if new_status == OrderStatus.CANCELLED else -1  # +1 вернуть на склад, -1 списать заново
        for it in order.items:
            stock = (
                await db.execute(
                    select(Stock)
                    .where(Stock.product_id == it.product_id, Stock.city_id == order.city_id)
                    .with_for_update()
                )
            ).scalar_one_or_none()
            if stock is None:
                stock = Stock(product_id=it.product_id, city_id=order.city_id, quantity=0)
                db.add(stock)
            if sign < 0 and stock.quantity < it.quantity:
                await db.rollback()
                raise OrderError("Нельзя восстановить заказ: товара на складе уже недостаточно")
            stock.quantity += sign * it.quantity
            db.add(InventoryTransaction(
                product_id=it.product_id, user_id=staff_id,
                type=TransactionType.RETURN if sign > 0 else TransactionType.SALE,
                quantity=sign * it.quantity,
                to_city_id=order.city_id if sign > 0 else None,
                from_city_id=order.city_id if sign < 0 else None,
                notes=f"{'Отмена' if sign > 0 else 'Восстановление'} заказа #{order.id}",
            ))

    order.status = new_status
    return order

"""
Модуль динамического ценообразования.
Берет коэффициенты из базы данных (SystemConfig) и рассчитывает цены.
"""
from decimal import Decimal, ROUND_HALF_UP
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import MobaPrice, SystemConfig

# Простой локальный кэш для коэффициентов, чтобы не ходить в БД на каждый запрос
_config_cache = {}

async def get_config_value(db: AsyncSession, key: str, default: str) -> str:
    """Получает значение настройки из БД или возвращает дефолт."""
    global _config_cache
    if key in _config_cache:
        return _config_cache[key]

    result = await db.execute(select(SystemConfig.value).where(SystemConfig.key == key))
    val = result.scalar_one_or_none()

    final_val = val if val is not None else default
    _config_cache[key] = final_val
    return final_val

def round_money(value: Decimal) -> Decimal:
    """Округляет до копеек по стандартным правилам."""
    return value.quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


def buyer_prices(mp, wholesale: bool) -> dict:
    """
    Цены, которые видит и платит конкретный покупатель (единое правило для витрины, корзины и заказа):
      price — наличными/переводом: розница our_base, оптовику our_opt (если рассчитана);
      card  — картой при получении: та же цена с наценкой за карту (our_card / our_base).
    Без our_base — «Цена по запросу» (обе None).
    """
    base = Decimal(mp.our_base) if mp.our_base is not None else None
    if base is None or base <= 0:
        return {"price": None, "card": None, "wholesale": False}
    opt = Decimal(mp.our_opt) if mp.our_opt else None
    card = Decimal(mp.our_card) if mp.our_card else base
    if wholesale and opt and opt > 0:
        return {"price": opt, "card": round_money(opt * card / base), "wholesale": True}
    return {"price": base, "card": card, "wholesale": False}

async def calculate_prices(db: AsyncSession, moba_price: MobaPrice) -> dict:
    """
    Рассчитывает наши цены на основе динамических коэффициентов из БД.
    """
    # Получаем коэффициенты из БД через кэш
    base_from = await get_config_value(db, "BASE_FROM", "regular")
    coef_base = Decimal(await get_config_value(db, "COEF_BASE", "0.9"))
    coef_card = Decimal(await get_config_value(db, "COEF_CARD", "1.1"))
    coef_opt = Decimal(await get_config_value(db, "COEF_OPT", "0.9"))

    if base_from == "discount":
        source = moba_price.discount_price or moba_price.price
    else:
        source = moba_price.price

    if source is None or source <= 0:
        return {"base": None, "card": None, "opt": None}

    source = Decimal(source)
    base = round_money(source * coef_base)
    card = round_money(base * coef_card)
    opt = round_money(base * coef_opt)

    return {"base": base, "card": card, "opt": opt}

async def recalculate_all(db: AsyncSession) -> int:
    """
    Пересчитывает цены товаров, загруженных парсером Мобы (у них заполнен url).
    Цены поставок (scripts/import_supply.py) и товаров, созданных в админке, задаются вручную —
    формула «цена Мобы × коэффициент» к ним не относится, их не трогаем.
    """
    records = await db.execute(select(MobaPrice).where(MobaPrice.url.isnot(None)))
    all_records = records.scalars().all()
    updated = 0

    for record in all_records:
        prices = await calculate_prices(db, record)
        record.our_base = prices["base"]
        record.our_card = prices["card"]
        record.our_opt = prices["opt"]
        updated += 1

    await db.commit()
    return updated

"""
Поиск товаров по ключевым словам (витрина, подсказки, приход/списание в админке).

Запрос разбивается на слова, каждое слово должно найтись в названии или артикуле (И):
- латиница и цифры ищутся как целое слово: «11» не находит «110», «c67» находит «C53/C67»;
- русские слова сводятся к основе («рамке» → «рамк») и расширяются синонимами («акб» → «аккумулятор»);
- предлоги («в», «для», «с» …) отбрасываются.
Ранжирование: точный артикул → артикул с префикса → название начинается с типа детали →
меньше «лишних» модификаторов модели (Pro, Max, Plus… которых нет в запросе) → короче название.

Пример: «iPhone 11 дисплей», «realme c67 акб», «honor 10i дисплей в рамке», «iPhone 13 pro max шлейф», «BTT-PMIG110-HC».
"""
import re

from sqlalchemy import and_, or_, case, exists, func, literal
from sqlalchemy.sql.elements import ColumnElement

from app.models import MobaPrice, Product, Stock

STOPWORDS = {"в", "во", "на", "для", "с", "со", "и", "из", "по", "к", "от", "под", "без"}

# Синонимы: основа слова в запросе → основы, которые ищем в названиях
SYNONYMS = {
    "акб": ["аккумулятор", "акб", "батаре"],
    "аккум": ["аккумулятор", "акб"],
    "батар": ["батаре", "аккумулятор"],
    "диспл": ["диспле", "экран"],
    "экран": ["диспле", "экран"],
    "модул": ["диспле", "модул"],
    "тач": ["тачскрин"],
    "тачскрин": ["тачскрин"],
    "крышк": ["крышк"],
    "корпус": ["корпус", "крышк"],
    "зарядк": ["заряд"],
    "заряд": ["заряд"],
    "разъем": ["разъем", "разъём"],
    "разъём": ["разъем", "разъём"],
    "стекл": ["стекл"],
    "шлейф": ["шлейф"],
    "камер": ["камер"],
    "динамик": ["динамик"],
    "рамк": ["рамк"],
    "наушник": ["наушник", "гарнитур"],
    "пауэрбанк": ["power bank", "внешн"],
    # бренды кириллицей
    "айфон": ["iphone"],
    "айпад": ["ipad"],
    "самсунг": ["samsung"],
    "ксиоми": ["xiaomi"],
    "сяоми": ["xiaomi"],
    "редми": ["redmi"],
    "хонор": ["honor"],
    "хуавей": ["huawei"],
    "реалми": ["realme"],
    "техно": ["tecno"],
    "инфиникс": ["infinix"],
    "поко": ["poco"],
}

# Модификаторы модели: если их нет в запросе, такие товары опускаются ниже
QUALIFIERS = ("pro", "max", "plus", "mini", "lite", "ultra", "se", "fe", "prime", "neo", "air")

ENDINGS = sorted(
    ("ами", "ями", "ого", "ему", "ыми", "ими", "ах", "ях", "ов", "ев", "ей", "ой", "ый", "ий", "ая", "яя",
     "ое", "ее", "ые", "ие", "ом", "ем", "ам", "ям", "а", "я", "ы", "и", "е", "у", "ю", "о", "ь"),
    key=len, reverse=True,
)

TOKEN_RE = re.compile(r"[0-9a-zа-яё]+", re.IGNORECASE)
ARTICLE_RE = re.compile(r"^[0-9A-Za-z][0-9A-Za-z\-_.]{2,}$")


def _stem(word: str) -> str:
    if len(word) <= 4:
        return word
    for e in ENDINGS:
        if word.endswith(e) and len(word) - len(e) >= 3:
            return word[: -len(e)]
    return word


def parse(query: str) -> dict:
    q = (query or "").strip()[:100]
    words = [w.lower().replace("ё", "е") for w in TOKEN_RE.findall(q)][:8]
    latin, cyr = [], []
    for w in words:
        if w in STOPWORDS:
            continue
        if re.fullmatch(r"[0-9a-z]+", w):
            latin.append(w)
        else:
            cyr.append(_stem(w))
    return {"raw": q, "latin": latin, "cyr": cyr, "article": q if ARTICLE_RE.match(q) and " " not in q else None}


def _cyr_condition(stem: str) -> ColumnElement:
    variants = SYNONYMS.get(stem) or next((v for k, v in SYNONYMS.items() if stem.startswith(k)), None) or [stem]
    conds = []
    for v in variants:
        if re.fullmatch(r"[0-9a-z ]+", v):
            conds.append(MobaPrice.name.op("~*")(r"\m" + re.escape(v).replace(r"\ ", " ") + r"\M"))
        else:
            conds.append(MobaPrice.name.ilike(f"%{v}%"))
    return or_(*conds)


def conditions(query: str) -> list[ColumnElement]:
    """Условия WHERE для MobaPrice. Пустой список — запрос пустой."""
    p = parse(query)
    conds = []
    for w in p["latin"]:
        conds.append(or_(MobaPrice.name.op("~*")(r"\m" + w + r"\M"), MobaPrice.sku.ilike(f"%{w}%")))
    for s in p["cyr"]:
        conds.append(_cyr_condition(s))
    if p["article"]:
        by_article = MobaPrice.sku.ilike(f"{p['article']}%")
        return [or_(by_article, and_(*conds))] if conds else [by_article]
    return conds


def score(query: str) -> ColumnElement:
    """Выражение релевантности (больше — выше в выдаче)."""
    p = parse(query)
    parts = [literal(0)]
    if p["article"]:
        parts.append(case((func.upper(MobaPrice.sku) == p["article"].upper(), 1000), else_=0))
        parts.append(case((MobaPrice.sku.ilike(f"{p['article']}%"), 500), else_=0))
    for i, s in enumerate(p["cyr"]):
        # «Дисплей для …» выше, чем «Наклейка на дисплей для …»; первое слово-тип в запросе главнее
        # («дисплей в рамке» — это дисплей, а не рамка)
        variants = SYNONYMS.get(s) or [s]
        parts.append(case((or_(*(MobaPrice.name.ilike(f"{v}%") for v in variants)), 40 if i == 0 else 15), else_=0))
    if len(p["latin"]) > 1:
        phrase = r"\m" + r"[\s/]+".join(p["latin"]) + r"\M"
        parts.append(case((MobaPrice.name.op("~*")(phrase), 30), else_=0))
    for qual in QUALIFIERS:
        if qual not in p["latin"]:
            parts.append(case((MobaPrice.name.op("~*")(r"\m" + qual + r"\M"), -25), else_=0))
    # Сначала то, что можно купить сейчас: с ценой и в наличии (иначе вверху «Цена по запросу» и «Нет в наличии»)
    parts.append(availability())
    expr = parts[0]
    for part in parts[1:]:
        expr = expr + part
    return expr


def availability() -> ColumnElement:
    """+300 — есть цена, +200 — есть на складе в любом городе."""
    in_stock = exists().where(Product.sku == MobaPrice.sku, Stock.product_id == Product.id, Stock.quantity > 0)
    return case((MobaPrice.our_base.isnot(None), 300), else_=0) + case((in_stock, 200), else_=0)


def order_by(query: str) -> list:
    return [score(query).desc(), func.length(MobaPrice.name), MobaPrice.id]

def money(value) -> str:
    """Как макрос money на сайте: «2 875,50 ₽», «208 ₽»."""
    v = Decimal(value or 0).quantize(Decimal("0.01"))
    if v != v.to_integral_value():
        s = "{:,.2f}".format(v).replace(",", " ").replace(".", ",")
    else:
        s = "{:,.0f}".format(v).replace(",", " ")
    return s + " ₽"


def keyboard(rows: list[list[tuple[str, str]]]) -> InlineKeyboardMarkup | None:
    """[[(текст, "url:…" | callback_data)]] → InlineKeyboardMarkup."""
    rows = [[(t, v) for t, v in row if not v.startswith("url:") or v.startswith("url:https://")] for row in rows or []]
    rows = [row for row in rows if row]
    if not rows:
        return None
    return InlineKeyboardMarkup(inline_keyboard=[
        [InlineKeyboardButton(text=t, url=v[4:]) if v.startswith("url:") else InlineKeyboardButton(text=t, callback_data=v)
         for t, v in row]
        for row in rows
    ])


async def send(chat_id: int, text: str, rows: list[list[tuple[str, str]]] | None = None) -> bool:

        return True
    except Exception:  # заблокировал бота, удалил аккаунт, сеть — не критично
        logger.warning("Telegram: не удалось отправить сообщение %s", chat_id, exc_info=True)
        return False


def fire(coro) -> None:

    task.add_done_callback(_tasks.discard)

def new_token(kind: str, value: str | int = "") -> str:
    token = secrets.token_urlsafe(18).replace("-", "x").replace("_", "y")  # deep-link: только [A-Za-z0-9]
    _redis().set(f"tg:{kind}:{token}", str(value), ex=TTL[kind])

    return _redis().get(f"tg:{kind}:{token}") if token else None

    if not token:

    return _redis().getdel(f"tg:{kind}:{token}")
